Q7. Create two integer stacks in a single array.


#include <stdio.h>

#define MAX 10

int stack[MAX];
int top1 = -1;
int top2 = MAX;

void push1(int value)
{
    if (top1 + 1 == top2)
        printf("Stack is Full\n");
    else
    {
        top1++;
        stack[top1] = value;
        printf("%d pushed into Stack 1\n", value);
    }
}

void push2(int value)
{
    if (top1 + 1 == top2)
        printf("Stack is Full\n");
    else
    {
        top2--;
        stack[top2] = value;
        printf("%d pushed into Stack 2\n", value);
    }
}

void pop1()
{
    if (top1 == -1)
        printf("Stack 1 is Empty\n");
    else
    {
        printf("%d popped from Stack 1\n", stack[top1]);
        top1--;
    }
}

void pop2()
{
    if (top2 == MAX)
        printf("Stack 2 is Empty\n");
    else
    {
        printf("%d popped from Stack 2\n", stack[top2]);
        top2++;
    }
}

int main()
{
    int n, i, value;

    printf("Enter number of elements for Stack 1: ");
    scanf("%d", &n);

    printf("Enter elements for Stack 1:\n");

    for (i = 0; i < n; i++)
    {
        scanf("%d", &value);
        push1(value);
    }

    printf("\nEnter number of elements for Stack 2: ");
    scanf("%d", &n);

    printf("Enter elements for Stack 2:\n");

    for (i = 0; i < n; i++)
    {
        scanf("%d", &value);
        push2(value);
    }

    printf("\n");

    pop1();
    pop2();

    return 0;
}
