Q12. Create a program to evaluate the postfix expression.


#include <stdio.h>
#include <string.h>

int stack[50], top = -1;

void push(int x)
{
    stack[++top] = x;
}

int pop()
{
    return stack[top--];
}

int main()
{
    char exp[50], ch;
    int i, a, b;

    printf("Enter postfix expression: ");
    scanf("%s", exp);

    for(i = 0; exp[i] != '\0'; i++)
    {
        ch = exp[i];

        if(ch >= '0' && ch <= '9')
            push(ch - '0');

        else
        {
            b = pop();
            a = pop();

            if(ch == '+')
                push(a + b);
            else if(ch == '-')
                push(a - b);
            else if(ch == '*')
                push(a * b);
            else if(ch == '/')
                push(a / b);
        }
    }

    printf("Result = %d", pop());

    return 0;
}

Output

-
Enter postfix expression: 23+
Result = 5
ANOTHER EXAMPLE:-
Enter postfix expression: 23*4+
Result = 10
