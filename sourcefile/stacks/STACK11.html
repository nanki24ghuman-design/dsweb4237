Q11. Create a program to evaluate the prefix expression.


#include <stdio.h>
#include <string.h>
#include <ctype.h>

int stack[50];
int top = -1;

void push(int n)
{
    top++;
    stack[top] = n;
}

int pop()
{
    int n = stack[top];
    top--;
    return n;
}

int main()
{
    char exp[50], ch;
    int i, a, b, result;

    printf("Enter prefix expression: ");
    scanf("%s", exp);

    for(i = strlen(exp) - 1; i >= 0; i--)
    {
        ch = exp[i];

        if(isdigit(ch))
        {
            push(ch - '0');
        }
        else
        {
            a = pop();
            b = pop();

            if(ch == '+')
                result = a + b;
            else if(ch == '-')
                result = a - b;
            else if(ch == '*')
                result = a * b;
            else if(ch == '/')
                result = a / b;

            push(result);
        }
    }

    printf("Result = %d", pop());

    return 0;
}

Output

-
Enter prefix expression: +23
Result = 5
ANOTHER EXAMPLE:-
Enter prefix expression: *+234
Result = 20
