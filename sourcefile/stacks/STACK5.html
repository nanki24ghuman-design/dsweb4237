Q5. Enter the string from the user and reverse the string using stacks.


#include <stdio.h>
#include <string.h>

#define MAX 100

char stack[MAX];
int top = -1;

int main()
{
    char str[MAX];
    int i;

    printf("Enter a string: ");
    scanf(" %[^\n]", str);

    for (i = 0; i < strlen(str); i++)
    {
        top++;
        stack[top] = str[i];
    }

    printf("Reversed string: ");

    while (top != -1)
    {
        printf("%c", stack[top]);
        top--;
    }

    printf("\n");

    return 0;
}

Output

-
Enter a string: HELLO WORLD
Reversed string: DLROW OLLEH
