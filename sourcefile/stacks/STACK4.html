Q4. Create a stack implementing the operations on the characters nodes (dynamically) (push, pop, peek, isempty, isfull)


#include <stdio.h>
#include <stdlib.h>

struct node
{
    char data;
    struct node *next;
};

struct node *top = NULL;

void push(char value)
{
    struct node *newnode;

    newnode = (struct node *)malloc(sizeof(struct node));

    if (newnode == NULL)
        printf("Stack is Full\n");
    else
    {
        newnode->data = value;
        newnode->next = top;
        top = newnode;

        printf("%c pushed into stack\n", value);
    }
}

void pop()
{
    struct node *temp;

    if (top == NULL)
        printf("Stack is Empty\n");
    else
    {
        temp = top;
        printf("%c popped from stack\n", top->data);
        top = top->next;
        free(temp);
    }
}

void peek()
{
    if (top == NULL)
        printf("Stack is Empty\n");
    else
        printf("Top element is %c\n", top->data);
}

void isEmpty()
{
    if (top == NULL)
        printf("Stack is Empty\n");
    else
        printf("Stack is Not Empty\n");
}

void isFull()
{
    struct node *temp;

    temp = (struct node *)malloc(sizeof(struct node));

    if (temp == NULL)
        printf("Stack is Full\n");
    else
    {
        printf("Stack is Not Full\n");
        free(temp);
    }
}

int main()
{
    int n, i;
    char value;

    printf("Enter number of elements: ");
    scanf("%d", &n);

    printf("Enter characters:\n");

    for (i = 0; i < n; i++)
    {
        scanf(" %c", &value);
        push(value);
    }

    peek();
    isEmpty();
    isFull();

    pop();
    peek();

    return 0;
}
