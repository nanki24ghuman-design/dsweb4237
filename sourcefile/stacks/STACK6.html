Q6. Create the program to match the single bracket using stacks.


#include <stdio.h>

#define MAX 100

char stack[MAX];
int top = -1;

int main()
{
    char expression[MAX];
    int i;
    int balanced = 1;

    printf("Enter expression: ");
    scanf(" %[^\n]", expression);

    for (i = 0; expression[i] != '\0'; i++)
    {
        if (expression[i] == '(')
        {
            top++;
            stack[top] = '(';
        }
        else if (expression[i] == ')')
        {
            if (top == -1)
            {
                balanced = 0;
                break;
            }

            top--;
        }
    }

    if (top != -1)
        balanced = 0;

    if (balanced == 1)
        printf("Brackets are Balanced\n");
    else
        printf("Brackets are Not Balanced\n");

    return 0;
}

-
Enter expression: (a+b)*(c+d)
Brackets are Balanced
