Q2. Create a stack implementing the operations on the characters array (push, pop, peek, isempty, isfull)


#include <stdio.h>

#define MAX 5

char stack[MAX];
int top = -1;

void push(char value)
{
    if (top == MAX - 1)
        printf("Stack is Full\n");
    else
    {
        top++;
        stack[top] = value;
        printf("%c pushed into stack\n", value);
    }
}

void pop()
{
    if (top == -1)
        printf("Stack is Empty\n");
    else
    {
        printf("%c popped from stack\n", stack[top]);
        top--;
    }
}

void peek()
{
    if (top == -1)
        printf("Stack is Empty\n");
    else
        printf("Top element is %c\n", stack[top]);
}

void isEmpty()
{
    if (top == -1)
        printf("Stack is Empty\n");
    else
        printf("Stack is Not Empty\n");
}

void isFull()
{
    if (top == MAX - 1)
        printf("Stack is Full\n");
    else
        printf("Stack is Not Full\n");
}

int main()
{
    int n, i;
    char value;

    printf("Enter number of elements: ");
    scanf("%d", &n);

    printf("Enter characters:\n");

    for (i = 0; i < n; i++)
    {
        scanf(" %c", &value);
        push(value);
    }

    peek();
    isEmpty();
    isFull();

    pop();
    peek();

    return 0;
}
