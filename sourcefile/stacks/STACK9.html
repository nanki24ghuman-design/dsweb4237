Q9. Enter the infix expression from user and convert into prefix expression.


#include <stdio.h>
#include <string.h>
#include <ctype.h>

char stack[100];
int top = -1;

void push(char ch)
{
    stack[++top] = ch;
}

char pop()
{
    return stack[top--];
}

int main()
{
    char infix[100], prefix[100], temp;
    int i, j = 0, len;

    printf("Enter infix expression: ");
    scanf("%s", infix);

    len = strlen(infix);

    for(i = len - 1; i >= 0; i--)
    {
        temp = infix[i];

        if(isalnum(temp))
        {
            prefix[j++] = temp;
        }
        else if(temp == ')')
        {
            push(temp);
        }
        else if(temp == '(')
        {
            while(top != -1 && stack[top] != ')')
                prefix[j++] = pop();

            if(top != -1)
                pop();
        }
        else
        {
            while(top != -1 && stack[top] != ')')
                prefix[j++] = pop();

            push(temp);
        }
    }

    while(top != -1)
        prefix[j++] = pop();

    prefix[j] = '\0';

    for(i = 0; i < j / 2; i++)
    {
        temp = prefix[i];
        prefix[i] = prefix[j - i - 1];
        prefix[j - i - 1] = temp;
    }

    printf("Prefix expression: %s", prefix);

    return 0;
}
