Q8. Sort the stack in ascending order using integer array.


#include <stdio.h>

#define MAX 100

int stack[MAX];
int top = -1;

int main()
{
    int n, i, j, temp;

    printf("Enter number of elements: ");
    scanf("%d", &n);

    printf("Enter elements:\n");

    for (i = 0; i < n; i++)
    {
        scanf("%d", &stack[i]);
        top++;
    }

    for (i = 0; i < n - 1; i++)
    {
        for (j = i + 1; j < n; j++)
        {
            if (stack[i] > stack[j])
            {
                temp = stack[i];
                stack[i] = stack[j];
                stack[j] = temp;
            }
        }
    }

    printf("Stack in ascending order:\n");

    for (i = 0; i < n; i++)
        printf("%d ", stack[i]);

    printf("\n");

    return 0;
}
