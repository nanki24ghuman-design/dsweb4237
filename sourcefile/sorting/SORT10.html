Q10. Enter a integer array from the user and perform quick sort algorithm through recursion. also calculate the worst and best complexity with exact time.


#include <stdio.h>

// Function to place the pivot at its correct position
int partition(int a[], int low, int high)
{
    int pivot, i, j, temp;

    pivot = a[high];
    i = low - 1;

    for (j = low; j < high; j++)
    {
        if (a[j] <= pivot)
        {
            i++;

            // Swap the elements
            temp = a[i];
            a[i] = a[j];
            a[j] = temp;
        }
    }

    // Place pivot at its correct position
    temp = a[i + 1];
    a[i + 1] = a[high];
    a[high] = temp;

    return i + 1;
}

// Function to perform quick sort using recursion
void quickSort(int a[], int low, int high)
{
    int pivotPosition;

    if (low < high)
    {
        // Find the pivot position
        pivotPosition = partition(a, low, high);

        // Sort the left part
        quickSort(a, low, pivotPosition - 1);

        // Sort the right part
        quickSort(a, pivotPosition + 1, high);
    }
}

int main()
{
    int a[100], n, i;

    printf("Enter the number of elements: ");
    scanf("%d", &n);

    printf("Enter array elements:\n");
    for (i = 0; i < n; i++)
    {
        scanf("%d", &a[i]);
    }

    // Call the sorting function
    quickSort(a, 0, n - 1);

    printf("Sorted array:\n");
    for (i = 0; i < n; i++)
    {
        printf("%d ", a[i]);
    }

    printf("\nBest Case Time Complexity = O(n log n)");
    printf("\nWorst Case Time Complexity = O(n^2)");

    return 0;
}
