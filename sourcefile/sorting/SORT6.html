Q6. Enter a integer array from the user and perform insertion sort algorithm through recursion. also calculate the worst and best complexity with exact time.


#include <stdio>

// Function to perform insertion sort using recursion
void insertionSort(int a[], int n)
{
    int key, j;

    // Stop when the array has one element
    if (n <= 1)
        return;

    // Sort the first n-1 elements
    insertionSort(a, n - 1);

    // Insert the last element at its correct position
    key = a[n - 1];
    j = n - 2;

    while (j >= 0 && a[j] > key)
    {
        a[j + 1] = a[j];
        j--;
    }

    a[j + 1] = key;
}

int main()
{
    int a[100], n, i;

    printf("Enter the number of elements: ");
    scanf("%d", &n);

    printf("Enter array elements:\n");
    for (i = 0; i < n; i++)
    {
        scanf("%d", &a[i]);
    }

    // Call the sorting function
    insertionSort(a, n);

    printf("Sorted array:\n");
    for (i = 0; i < n; i++)
    {
        printf("%d ", a[i]);
    }

    printf("\nBest Case Time Complexity = O(n)");
    printf("\nWorst Case Time Complexity = O(n^2)");

    return 0;
}
