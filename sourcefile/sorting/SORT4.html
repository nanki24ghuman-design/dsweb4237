Q4. Enter a integer array from the user and perform bubble sort algorithm through recursion. also calculate the worst and best complexity with exact time.


#include <stdio.h>

// Function to perform bubble sort using recursion
void bubbleSort(int a[], int n)
{
    int i, temp, swapped = 0;

    // Stop when the array has one element
    if (n == 1)
        return;

    // Compare adjacent elements
    for (i = 0; i < n - 1; i++)
    {
        if (a[i] > a[i + 1])
        {
            // Swap the elements
            temp = a[i];
            a[i] = a[i + 1];
            a[i + 1] = temp;

            swapped = 1;
        }
    }

    // Stop if the array is already sorted
    if (swapped == 0)
        return;

    // Call the function again
    bubbleSort(a, n - 1);
}

int main()
{
    int a[100], n, i;

    printf("Enter the number of elements: ");
    scanf("%d", &n);

    printf("Enter array elements:\n");
    for (i = 0; i < n; i++)
    {
        scanf("%d", &a[i]);
    }

    // Call the sorting function
    bubbleSort(a, n);

    printf("Sorted array:\n");
    for (i = 0; i < n; i++)
    {
        printf("%d ", a[i]);
    }

    printf("\nBest Case Time Complexity = O(n)");
    printf("\nWorst Case Time Complexity = O(n^2)");

    return 0;
}
