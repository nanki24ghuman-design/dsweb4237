Q1. Enter a integer array from the user and perform selection sort algorithm through iteration. also calculate the worst and best complexity with exact time.


#include<stdio.h>

int main()
{
    int array[100], size, i, j, minimum, temp;
    int comparisons = 0;

    printf("Enter the number of elements: ");
    scanf("%d", &size);

    printf("Enter array elements:\n");
    for (i = 0; i < size; i++)
    {
        scanf("%d", &array[i]);
    }

    // Selection Sort using iteration
    for (i = 0; i < size - 1; i++)
    {
        minimum = i;

        for (j = i + 1; j < size; j++)
        {
            comparisons++;

            if (array[j] < array[minimum])
            {
                minimum = j;
            }
        }

        // Swap the elements
        temp = array[i];
        array[i] = array[minimum];
        array[minimum] = temp;
    }

    printf("Sorted array in ascending order:\n");
    for (i = 0; i < size; i++)
    {
        printf("%d ", array[i]);
    }

    printf("\nTotal comparisons = %d", comparisons);

    printf("\nBest Case Time Complexity = O(n^2)");
    printf("\nWorst Case Time Complexity = O(n^2)");

    return 0;
}
