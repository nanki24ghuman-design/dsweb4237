Q16. Enter a string array from the user and perform insertion sort algorithm through recursion. also calculate the worst and best complexity with exact time.


#include <stdio.h>
#include <string.h>

// Function to perform insertion sort using recursion
int insertionSort(char a[][50], int n)
{
    int j, comparisons = 0;
    char key[50];

    // Base case
    if (n <= 1)
        return 0;

    // Sort the first n-1 strings
    comparisons = insertionSort(a, n - 1);

    // Take the last string as the key
    strcpy(key, a[n - 1]);
    j = n - 2;

    // Move larger strings one position ahead
    while (j >= 0)
    {
        comparisons++;

        if (strcmp(a[j], key) > 0)
        {
            strcpy(a[j + 1], a[j]);
            j--;
        }
        else
        {
            break;
        }
    }

    // Place the key in its correct position
    strcpy(a[j + 1], key);

    return comparisons;
}

int main()
{
    char a[100][50];
    int n, i, comparisons;

    printf("Enter the number of strings: ");
    scanf("%d", &n);

    printf("Enter strings:\n");

    for (i = 0; i < n; i++)
    {
        scanf("%49s", a[i]);
    }

    comparisons = insertionSort(a, n);

    printf("\nSorted strings:\n");

    for (i = 0; i < n; i++)
    {
        printf("%s\n", a[i]);
    }

    printf("\nTotal comparisons = %d", comparisons);
    printf("\nBest Case Time Complexity = O(n)");
    printf("\nWorst Case Time Complexity = O(n^2)");

    return 0;
}
