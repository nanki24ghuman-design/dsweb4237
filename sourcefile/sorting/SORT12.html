Q12. Enter a string array from the user and perform selection sort algorithm through recursion. also calculate the worst and best complexity with exact time.


#include <stdio.h>
#include <string.h>

// Function to perform selection sort using recursion
int selectionSort(char a[][50], int n, int index)
{
    int j, min;
    char temp[50];
    int comparisons = 0;

    if (index == n - 1)
        return 0;

    min = index;

    // Find the alphabetically smallest string
    for (j = index + 1; j < n; j++)
    {
        comparisons++;

        if (strcmp(a[j], a[min]) < 0)
        {
            min = j;
        }
    }

    // Swap the strings
    strcpy(temp, a[index]);
    strcpy(a[index], a[min]);
    strcpy(a[min], temp);

    // Sort the remaining strings recursively
    return comparisons + selectionSort(a, n, index + 1);
}

int main()
{
    char a[100][50];
    int n, i, comparisons;

    printf("Enter the number of strings: ");
    scanf("%d", &n);

    printf("Enter strings:\n");

    for (i = 0; i < n; i++)
    {
        scanf("%49s", a[i]);
    }

    comparisons = selectionSort(a, n, 0);

    printf("\nSorted strings:\n");

    for (i = 0; i < n; i++)
    {
        printf("%s\n", a[i]);
    }

    printf("\nTotal comparisons = %d", comparisons);
    printf("\nBest Case Time Complexity = O(n^2)");
    printf("\nWorst Case Time Complexity = O(n^2)");

    return 0;
}
