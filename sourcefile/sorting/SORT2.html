Q2. Enter a integer array from the user and perform selection sort algorithm through recursion. also calculate the worst and best complexity with exact time.


#include <stdio.h>
void selectionSort(int a[], int n, int index)
{
    int j, min, temp;

    // Stop when the array is sorted
    if (index == n - 1)
        return;

    min = index;

    // Find the smallest element
    for (j = index + 1; j < n; j++)
    {
        if (a[j] < a[min])
        {
            min = j;
        }
    }

    // Swap the elements
    temp = a[index];
    a[index] = a[min];
    a[min] = temp;

    // Call the function again
    selectionSort(a, n, index + 1);
}

int main()
{
    int a[100], n, i;

    printf("Enter the number of elements: ");
    scanf("%d", &n);

    printf("Enter array elements:\n");
    for (i = 0; i < n; i++)
    {
        scanf("%d", &a[i]);
    }

    // Call selection sort function
    selectionSort(a, n, 0);

    printf("Sorted array:\n");
    for (i = 0; i < n; i++)
    {
        printf("%d ", a[i]);
    }

    printf("\nBest Case Time Complexity = O(n^2)");
    printf("\nWorst Case Time Complexity = O(n^2)");

    return 0;
}
