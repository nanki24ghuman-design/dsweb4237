Q20. Enter a string array from the user and perform quick sort algorithm through recursion. also calculate the worst and best complexity with exact time.


#include <stdio.h>
#include <string.h>

// Function to arrange strings around the pivot
int partition(char a[][50], int low, int high, int *count)
{
    char pivot[50], temp[50];
    int i = low - 1, j;

    strcpy(pivot, a[high]);

    for (j = low; j < high; j++)
    {
        (*count)++;

        if (strcmp(a[j], pivot) < 0)
        {
            i++;

            strcpy(temp, a[i]);
            strcpy(a[i], a[j]);
            strcpy(a[j], temp);
        }
    }

    strcpy(temp, a[i + 1]);
    strcpy(a[i + 1], a[high]);
    strcpy(a[high], temp);

    return i + 1;
}

// Function to perform quick sort using recursion
void quickSort(char a[][50], int low, int high, int *count)
{
    int pivot;

    if (low < high)
    {
        pivot = partition(a, low, high, count);

        quickSort(a, low, pivot - 1, count);
        quickSort(a, pivot + 1, high, count);
    }
}

int main()
{
    char a[100][50];
    int n, i, count = 0;

    printf("Enter the number of strings: ");
    scanf("%d", &n);

    if (n < 1 || n > 100)
    {
        printf("Enter between 1 and 100 strings.");
        return 0;
    }

    printf("Enter strings:\n");

    for (i = 0; i < n; i++)
    {
        scanf("%49s", a[i]);
    }

    quickSort(a, 0, n - 1, &count);

    printf("\nSorted strings:\n");

    for (i = 0; i < n; i++)
    {
        printf("%s\n", a[i]);
    }

    printf("\nTotal comparisons = %d", count);
    printf("\nBest Case Time Complexity = O(n log n)");
    printf("\nWorst Case Time Complexity = O(n^2)");

    return 0;
}
