Q11. Enter a string array from the user and perform selection sort algorithm through iteration. also calculate the worst and best complexity with exact time.


#include <stdio.h>
#include <string.h>

// Function to perform selection sort
void selectionSort(char a[][50], int n)
{
    int i, j, min;
    char temp[50];

    for (i = 0; i < n - 1; i++)
    {
        min = i;

        // Find the alphabetically smallest string
        for (j = i + 1; j < n; j++)
        {
            if (strcmp(a[j], a[min]) < 0)
            {
                min = j;
            }
        }

        // Swap the strings
        strcpy(temp, a[i]);
        strcpy(a[i], a[min]);
        strcpy(a[min], temp);
    }
}

int main()
{
    char a[100][50];
    int n, i;

    printf("Enter the number of strings: ");
    scanf("%d", &n);

    // Input strings
    printf("Enter strings:\n");
    for (i = 0; i < n; i++)
    {
        scanf("%49s", a[i]);
    }

    // Call the sorting function
    selectionSort(a, n);

    // Display sorted strings
    printf("Sorted strings:\n");
    for (i = 0; i < n; i++)
    {
        printf("%s\n", a[i]);
    }

    printf("Best Case Time Complexity = O(n^2)");
    printf("\nWorst Case Time Complexity = O(n^2)");

    return 0;
}
