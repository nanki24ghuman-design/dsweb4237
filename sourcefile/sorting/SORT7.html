Q7. Enter a integer array from the user and perform merge sort algorithm through iteration. also calculate the worst and best complexity with exact time.


#include <stdio.h>

// Function to merge two sorted parts
void merge(int a[], int temp[], int left, int mid, int right)
{
    int i = left, j = mid + 1, k = left;

    while (i <= mid && j <= right)
    {
        if (a[i] <= a[j])
            temp[k++] = a[i++];
        else
            temp[k++] = a[j++];
    }

    // Copy remaining elements
    while (i <= mid)
        temp[k++] = a[i++];

    while (j <= right)
        temp[k++] = a[j++];

    // Copy elements back to original array
    for (i = left; i <= right; i++)
        a[i] = temp[i];
}

// Function to perform merge sort using iteration
void mergeSort(int a[], int n)
{
    int temp[100];
    int size, left, mid, right;

    // Increase the size of parts in each pass
    for (size = 1; size < n; size = size * 2)
    {
        for (left = 0; left < n - size; left = right + 1)
        {
            mid = left + size - 1;
            right = left + 2 * size - 1;

            if (right >= n)
                right = n - 1;

            // Merge the two parts
            merge(a, temp, left, mid, right);
        }
    }
}

int main()
{
    int a[100], n, i;

    printf("Enter the number of elements: ");
    scanf("%d", &n);

    printf("Enter array elements:\n");
    for (i = 0; i < n; i++)
    {
        scanf("%d", &a[i]);
    }

    // Call merge sort function
    mergeSort(a, n);

    printf("Sorted array:\n");
    for (i = 0; i < n; i++)
    {
        printf("%d ", a[i]);
    }

    printf("\nBest Case Time Complexity = O(n log n)");
    printf("\nWorst Case Time Complexity = O(n log n)");

    return 0;
}
