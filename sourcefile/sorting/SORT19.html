Q19. Enter a string array from the user and perform quick sort algorithm through iteration. also calculate the worst and best complexity with exact time.

#include <stdio.h>
#include <string.h>

// Function to arrange strings around the pivot
int partition(char a[][50], int low, int high, int *comparisons)
{
    char pivot[50], temp[50];
    int i = low - 1, j;

    strcpy(pivot, a[high]);

    for (j = low; j < high; j++)
    {
        (*comparisons)++;

        if (strcmp(a[j], pivot) < 0)
        {
            i++;

            strcpy(temp, a[i]);
            strcpy(a[i], a[j]);
            strcpy(a[j], temp);
        }
    }

    // Place the pivot in its correct position
    strcpy(temp, a[i + 1]);
    strcpy(a[i + 1], a[high]);
    strcpy(a[high], temp);

    return i + 1;
}

// Function to perform quick sort using iteration
int quickSort(char a[][50], int n)
{
    int stack[200];
    int top = -1;
    int low, high, pivot;
    int comparisons = 0;

    if (n <= 1)
        return 0;

    // Store the starting and ending indexes
    stack[++top] = 0;
    stack[++top] = n - 1;

    while (top >= 0)
    {
        high = stack[top--];
        low = stack[top--];

        if (low < high)
        {
            pivot = partition(a, low, high, &comparisons);

            // Store the left part
            if (pivot - 1 > low)
            {
                stack[++top] = low;
                stack[++top] = pivot - 1;
            }

            // Store the right part
            if (pivot + 1 < high)
            {
                stack[++top] = pivot + 1;
                stack[++top] = high;
            }
        }
    }

    return comparisons;
}

int main()
{
    char a[100][50];
    int n, i, comparisons;

    printf("Enter the number of strings: ");
    scanf("%d", &n);

    if (n < 1 || n > 100)
    {
        printf("Please enter between 1 and 100 strings.");
        return 0;
    }

    printf("Enter strings:\n");

    for (i = 0; i < n; i++)
    {
        scanf("%49s", a[i]);
    }

    comparisons = quickSort(a, n);

    printf("\nSorted strings:\n");

    for (i = 0; i < n; i++)
    {
        printf("%s\n", a[i]);
    }

    printf("\nTotal comparisons = %d", comparisons);
    printf("\nBest Case Time Complexity = O(n log n)");
    printf("\nWorst Case Time Complexity = O(n^2)");

    return 0;
}
