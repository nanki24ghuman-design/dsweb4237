Q17. Enter a string array from the user and perform merge sort algorithm through iteration. also calculate the worst and best complexity with exact time.


#include <stdio.h>
#include <string.h>

// Function to merge two sorted parts
int merge(char a[][50], char temp[][50], int left, int mid, int right)
{
    int i = left, j = mid + 1, k = left;
    int comparisons = 0;

    while (i <= mid && j <= right)
    {
        comparisons++;

        if (strcmp(a[i], a[j]) <= 0)
        {
            strcpy(temp[k], a[i]);
            i++;
        }
        else
        {
            strcpy(temp[k], a[j]);
            j++;
        }
        k++;
    }

    while (i <= mid)
    {
        strcpy(temp[k], a[i]);
        i++;
        k++;
    }

    while (j <= right)
    {
        strcpy(temp[k], a[j]);
        j++;
        k++;
    }

    for (i = left; i <= right; i++)
    {
        strcpy(a[i], temp[i]);
    }

    return comparisons;
}

// Function to perform iterative merge sort
int mergeSort(char a[][50], int n)
{
    char temp[100][50];
    int size, left, mid, right, i;
    int comparisons = 0;

    for (size = 1; size < n; size = size * 2)
    {
        for (left = 0; left < n - size; left = left + 2 * size)
        {
            mid = left + size - 1;
            right = left + 2 * size - 1;

            if (right >= n)
                right = n - 1;

            comparisons += merge(a, temp, left, mid, right);
        }
    }

    return comparisons;
}

int main()
{
    char a[100][50];
    int n, i, comparisons;

    printf("Enter the number of strings: ");
    scanf("%d", &n);

    printf("Enter strings:\n");

    for (i = 0; i < n; i++)
    {
        scanf("%49s", a[i]);
    }

    comparisons = mergeSort(a, n);

    printf("\nSorted strings:\n");

    for (i = 0; i < n; i++)
    {
        printf("%s\n", a[i]);
    }

    printf("\nTotal comparisons = %d", comparisons);
    printf("\nBest Case Time Complexity = O(n log n)");
    printf("\nWorst Case Time Complexity = O(n log n)");

    return 0;
}
