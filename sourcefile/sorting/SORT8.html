Q8. Enter a integer array from the user and perform merge sort algorithm through recursion. also calculate the worst and best complexity with exact time.


#include <stdio.h>

// Function to merge two sorted parts
void merge(int a[], int left, int mid, int right)
{
    int temp[100];
    int i = left, j = mid + 1, k = left;

    // Compare and merge elements
    while (i <= mid && j <= right)
    {
        if (a[i] <= a[j])
        {
            temp[k] = a[i];
            i++;
        }
        else
        {
            temp[k] = a[j];
            j++;
        }
        k++;
    }

    // Copy remaining elements
    while (i <= mid)
    {
        temp[k] = a[i];
        i++;
        k++;
    }

    while (j <= right)
    {
        temp[k] = a[j];
        j++;
        k++;
    }

    // Copy elements back to the original array
    for (i = left; i <= right; i++)
    {
        a[i] = temp[i];
    }
}

// Function to perform merge sort using recursion
void mergeSort(int a[], int left, int right)
{
    int mid;

    if (left < right)
    {
        mid = (left + right) / 2;

        // Sort the left part
        mergeSort(a, left, mid);

        // Sort the right part
        mergeSort(a, mid + 1, right);

        // Merge both sorted parts
        merge(a, left, mid, right);
    }
}

int main()
{
    int a[100], n, i;

    printf("Enter the number of elements: ");
    scanf("%d", &n);

    printf("Enter array elements:\n");
    for (i = 0; i < n; i++)
    {
        scanf("%d", &a[i]);
    }

    // Call the sorting function
    mergeSort(a, 0, n - 1);

    printf("Sorted array:\n");
    for (i = 0; i < n; i++)
    {
        printf("%d ", a[i]);
    }

    printf("\nBest Case Time Complexity = O(n log n)");
    printf("\nWorst Case Time Complexity = O(n log n)");

    return 0;
}
