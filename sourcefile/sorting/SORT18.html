Q18. Enter a string array from the user and perform merge sort algorithm through recursion. also calculate the worst and best complexity with exact time.


#include <stdio.h>
#include <string.h>

// Function to merge two sorted parts
int merge(char a[][50], int left, int mid, int right)
{
    char temp[100][50];
    int i = left, j = mid + 1, k = left;
    int comparisons = 0;

    // Compare strings and merge them in alphabetical order
    while (i <= mid && j <= right)
    {
        comparisons++;

        if (strcmp(a[i], a[j]) <= 0)
        {
            strcpy(temp[k], a[i]);
            i++;
        }
        else
        {
            strcpy(temp[k], a[j]);
            j++;
        }

        k++;
    }

    // Copy remaining strings from the left part
    while (i <= mid)
    {
        strcpy(temp[k], a[i]);
        i++;
        k++;
    }

    // Copy remaining strings from the right part
    while (j <= right)
    {
        strcpy(temp[k], a[j]);
        j++;
        k++;
    }

    // Copy merged strings back to the original array
    for (i = left; i <= right; i++)
    {
        strcpy(a[i], temp[i]);
    }

    return comparisons;
}

// Function to perform merge sort using recursion
int mergeSort(char a[][50], int left, int right)
{
    int mid, comparisons = 0;

    if (left < right)
    {
        mid = (left + right) / 2;

        // Sort the left part
        comparisons += mergeSort(a, left, mid);

        // Sort the right part
        comparisons += mergeSort(a, mid + 1, right);

        // Merge both sorted parts
        comparisons += merge(a, left, mid, right);
    }

    return comparisons;
}

int main()
{
    char a[100][50];
    int n, i, comparisons;

    printf("Enter the number of strings: ");
    scanf("%d", &n);

    printf("Enter strings:\n");

    for (i = 0; i < n; i++)
    {
        scanf("%49s", a[i]);
    }

    comparisons = mergeSort(a, 0, n - 1);

    printf("\nSorted strings:\n");

    for (i = 0; i < n; i++)
    {
        printf("%s\n", a[i]);
    }

    printf("\nTotal comparisons = %d", comparisons);
    printf("\nBest Case Time Complexity = O(n log n)");
    printf("\nWorst Case Time Complexity = O(n log n)");

    return 0;
}
