Q9. Enter a integer array from the user and perform quick sort algorithm through iteration. also calculate the worst and best complexity with exact time.


#include<stdio.h>
// Function to partition the array
int partition(int a[], int low, int high)
{
    int pivot, i, j, temp;

    pivot = a[high];
    i = low - 1;

    for (j = low; j < high; j++)
    {
        if (a[j] <= pivot)
        {
            i++;

            // Swap the elements
            temp = a[i];
            a[i] = a[j];
            a[j] = temp;
        }
    }

    // Place pivot at its correct position
    temp = a[i + 1];
    a[i + 1] = a[high];
    a[high] = temp;

    return i + 1;
}

// Function to perform quick sort using iteration
void quickSort(int a[], int n)
{
    int stack[100], top = -1;
    int low, high, pivotPosition;

    // Store the starting and ending positions
    stack[++top] = 0;
    stack[++top] = n - 1;

    while (top >= 0)
    {
        high = stack[top--];
        low = stack[top--];

        pivotPosition = partition(a, low, high);

        // Store the left part
        if (pivotPosition - 1 > low)
        {
            stack[++top] = low;
            stack[++top] = pivotPosition - 1;
        }

        // Store the right part
        if (pivotPosition + 1 < high)
        {
            stack[++top] = pivotPosition + 1;
            stack[++top] = high;
        }
    }
}

int main()
{
    int a[100], n, i;

    printf("Enter the number of elements: ");
    scanf("%d", &n);

    printf("Enter array elements:\n");
    for (i = 0; i < n; i++)
    {
        scanf("%d", &a[i]);
    }

    // Call the sorting function
    quickSort(a, n);

    printf("Sorted array:\n");
    for (i = 0; i < n; i++)
    {
        printf("%d ", a[i]);
    }

    printf("\nBest Case Time Complexity = O(n log n)");
    printf("\nWorst Case Time Complexity = O(n^2)");

    return 0;
}
