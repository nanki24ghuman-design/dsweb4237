Q3. Enter a integer array from the user and perform bubble sort algorithm through iteration. also calculate the worst and best complexity with exact time.


#include <stdio.h>

// Function to perform bubble sort
void bubbleSort(int a[], int n)
{
    int i, j, temp, swapped;

    for (i = 0; i < n - 1; i++)
    {
        swapped = 0;

        for (j = 0; j < n - 1 - i; j++)
        {
            if (a[j] > a[j + 1])
            {
                // Swap adjacent elements
                temp = a[j];
                a[j] = a[j + 1];
                a[j + 1] = temp;

                swapped = 1;
            }
        }

        // Stop if the array is already sorted
        if (swapped == 0)
            break;
    }
}

int main()
{
    int a[100], n, i;

    printf("Enter the number of elements: ");
    scanf("%d", &n);

    for (i = 0; i < n; i++)
    {
        scanf("%d", &a[i]);
    }

    // Call the sorting function
    bubbleSort(a, n);

    printf("Sorted array:\n");
    for (i = 0; i < n; i++)
    {
        printf("%d ", a[i]);
    }

    printf("\nBest Case Time Complexity = O(n)");
    printf("\nWorst Case Time Complexity = O(n^2)");

    return 0;
}
