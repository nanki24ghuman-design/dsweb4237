Q14. Enter a string array from the user and perform bubble sort algorithm through recursion. also calculate the worst and best complexity with exact time.


#include <stdio.h>
#include <string.h>

// Function to perform bubble sort using recursion
int bubbleSort(char a[][50], int n)
{
    int j, comparisons = 0, swapped = 0;
    char temp[50];

    if (n == 1)
        return 0;

    // Compare adjacent strings
    for (j = 0; j < n - 1; j++)
    {
        comparisons++;

        if (strcmp(a[j], a[j + 1]) > 0)
        {
            // Swap the strings
            strcpy(temp, a[j]);
            strcpy(a[j], a[j + 1]);
            strcpy(a[j + 1], temp);

            swapped = 1;
        }
    }

    // Stop if the strings are already sorted
    if (swapped == 0)
        return comparisons;

    // Sort the remaining strings using recursion
    return comparisons + bubbleSort(a, n - 1);
}

int main()
{
    char a[100][50];
    int n, i, comparisons;

    printf("Enter the number of strings: ");
    scanf("%d", &n);

    printf("Enter strings:\n");

    for (i = 0; i < n; i++)
    {
        scanf("%49s", a[i]);
    }

    comparisons = bubbleSort(a, n);

    printf("\nSorted strings:\n");

    for (i = 0; i < n; i++)
    {
        printf("%s\n", a[i]);
    }

    printf("\nTotal comparisons = %d", comparisons);
    printf("\nBest Case Time Complexity = O(n)");
    printf("\nWorst Case Time Complexity = O(n^2)");

    return 0;
  }
