Q3. Enter a integer array from the user, enter an integer value and perform linear search algorithm through recursion.


#include <stdio.h>

int search(int a[], int n, int value, int i)
{
    if(i == n)
        return -1;

    if(a[i] == value)
        return i;

    return search(a, n, value, i + 1);
}


int main()
{
    int a[100], n, value, i, result;

    printf("Enter size of array: ");
    scanf("%d", &n);

    printf("Enter array elements:\n");
    for(i = 0; i < n; i++)
    {
        scanf("%d", &a[i]);
    }

    printf("Enter value to search: ");
    scanf("%d", &value);

    result = search(a, n, value, 0);

    if(result == -1)
        printf("Value not found");
    else
        printf("Value found at position %d", result + 1);

    return 0;
}
