Q6. Enter a string from the user, enter a character and perform binary search algorithm through recursion.


#include <stdio.h>

int search(char str[], char ch, int low, int high)
{
    int mid;

    if(low > high)
        return -1;

    mid = (low + high) / 2;

    if(str[mid] == ch)
        return mid;
    else if(ch > str[mid])
        return search(str, ch, mid + 1, high);
    else
        return search(str, ch, low, mid - 1);
}

int main()
{
    char str[100], ch;
    int result, i;

    printf("Enter a sorted string: ");
    scanf("%s", str);

    printf("Enter character to search: ");
    scanf(" %c", &ch);

    i = 0;
    while(str[i] != '\0')
    {
        i++;
    }

    result = search(str, ch, 0, i - 1);

    if(result == -1)
        printf("Character not found");
    else
        printf("Character found at position %d", result + 1);

    return 0;
}
