Q8. Enter a string from the user, enter a character and perform binary search algorithm through iteration.


#include <stdio.h>

int main()
{
    char str[100], ch;
    int low, high, mid, i, found = 0;

    printf("Enter a sorted string: ");
    scanf("%s", str);

    printf("Enter character to search: ");
    scanf(" %c", &ch);

    i = 0;
    while(str[i] != '\0')
    {
        i++;
    }

    low = 0;
    high = i - 1;

    while(low <= high)
    {
        mid = (low + high) / 2;

        if(str[mid] == ch)
        {
            printf("Character found at position %d", mid + 1);
            found = 1;
            break;
        }
        else if(ch > str[mid])
        {
            low = mid + 1;
        }
        else
        {
            high = mid - 1;
        }
    }

    if(found == 0)
    {
        printf("Character not found");
    }

    return 0;
}
