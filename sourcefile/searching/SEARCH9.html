Q9. Calculate the best and worst complexity of the linear searching algorithm with exact time taken by the algorithm on integer array.


#include <stdio.h>

int main()
{
    int a[100], n, i, value, count;

    printf("Enter size of array: ");
    scanf("%d", &n);

    printf("Enter array elements:\n");
    for(i = 0; i < n; i++)
    {
        scanf("%d", &a[i]);
    }

    printf("Enter value to search: ");
    scanf("%d", &value);

    count = 0;

    for(i = 0; i < n; i++)
    {
        count++;

        if(a[i] == value)
        {
            printf("Value found at position %d\n", i + 1);
            break;
        }
    }

    if(i == n)
        printf("Value not found\n");

    printf("Number of comparisons: %d\n", count);

    if(count == 1)
        printf("Best case complexity: O(1)\n");
    else if(count == n)
        printf("Worst case complexity: O(n)\n");
    else
        printf("Average case complexity: O(n)\n");

    return 0;
}
