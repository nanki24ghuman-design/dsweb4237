Q5. Enter a integer array from the user, enter an integer value and perform binary search algorithm through iteration.


#include <stdio.h>

int main()
{
    int a[100], n, i, value;
    int low, high, mid, found = 0;

    printf("Enter size of array: ");
    scanf("%d", &n);

    printf("Enter elements in ascending order:\n");
    for(i = 0; i < n; i++)
    {
        scanf("%d", &a[i]);
    }

    printf("Enter value to search: ");
    scanf("%d", &value);

    low = 0;
    high = n - 1;

    while(low <= high)
    {
        mid = (low + high) / 2;

        if(a[mid] == value)
        {
            printf("Value found at position %d", mid + 1);
            found = 1;
            break;
        }
        else if(value > a[mid])
        {
            low = mid + 1;
        }
        else
        {
            high = mid - 1;
        }
    }

    if(found == 0)
    {
        printf("Value not found");
    }

    return 0;
}

Enter value to search: 40
