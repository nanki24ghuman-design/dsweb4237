Q2. Enter a string from the user, enter a character and perform linear search algorithm through recursion.


#include <stdio.h>
int search(char str[], char ch, int i)
{
    if(str[i] == '\0')
        return -1;

    if(str[i] == ch)
        return i;

    return search(str, ch, i + 1);
}
int main()
{
    char str[100], ch;
    int i, result;

    printf("Enter a string: ");
    scanf("%s", str);

    printf("Enter a character to search: ");
    scanf(" %c", &ch);

    result = search(str, ch, 0);

    if(result == -1)
        printf("Character not found");
    else
        printf("Character found at position %d", result + 1);

    return 0;
}
