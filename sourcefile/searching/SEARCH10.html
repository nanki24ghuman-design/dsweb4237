Q10. Calculate the best and worst complexity of the binary searching algorithm with exact time taken by the algorithm on integer array.


#include <stdio.h>

int main()
{
    int a[100], n, i, value;
    int low, high, mid, count = 0;

    printf("Enter size of array: ");
    scanf("%d", &n);

    printf("Enter elements in ascending order:\n");
    for(i = 0; i < n; i++)
    {
        scanf("%d", &a[i]);
    }

    printf("Enter value to search: ");
    scanf("%d", &value);

    low = 0;
    high = n - 1;

    while(low <= high)
    {
        mid = (low + high) / 2;
        count++;

        if(a[mid] == value)
        {
            printf("Value found at position %d\n", mid + 1);
            break;
        }
        else if(value > a[mid])
        {
            low = mid + 1;
        }
        else
        {
            high = mid - 1;
        }
    }

    if(low > high)
        printf("Value not found\n");

    printf("Number of comparisons: %d\n", count);

    if(count == 1)
        printf("Best case complexity: O(1)\n");
    else
        printf("Worst case complexity: O(log n)\n");

    return 0;
}
