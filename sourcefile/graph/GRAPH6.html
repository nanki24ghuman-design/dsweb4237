Q6. Traverse the adjacent list with breadth first search


#include <stdio.h>
#include <stdlib.h>

struct node
{
    int vertex;
    struct node *next;
};

struct node *a[10];
int visited[10];
int queue[10];

void BFS(int start)
{
    int front = 0, rear = 0;
    int current;
    struct node *temp;

    visited[start] = 1;
    queue[rear] = start;
    rear++;

    while (front < rear)
    {
        current = queue[front];
        front++;

        printf("%d ", current);

        temp = a[current];

        while (temp != NULL)
        {
            if (visited[temp->vertex] == 0)
            {
                visited[temp->vertex] = 1;
                queue[rear] = temp->vertex;
                rear++;
            }

            temp = temp->next;
        }
    }
}

int main()
{
    int v, e, i, x, y, start;
    struct node *temp;

    printf("Enter number of vertices: ");
    scanf("%d", &v);

    printf("Enter number of edges: ");
    scanf("%d", &e);

    for (i = 0; i < v; i++)
    {
        a[i] = NULL;
        visited[i] = 0;
    }

    printf("Enter edges (two vertices):\n");

    for (i = 0; i < e; i++)
    {
        scanf("%d %d", &x, &y);

        temp = (struct node *)malloc(sizeof(struct node));
        temp->vertex = y;
        temp->next = a[x];
        a[x] = temp;

        temp = (struct node *)malloc(sizeof(struct node));
        temp->vertex = x;
        temp->next = a[y];
        a[y] = temp;
    }

    printf("Enter starting vertex: ");
    scanf("%d", &start);

    printf("BFS Traversal: ");
    BFS(start);

    return 0;
}
