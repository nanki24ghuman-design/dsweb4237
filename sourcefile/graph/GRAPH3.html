Q3. Traverse the adjacent matrix with depth first search


#include <stdio.h>

int a[10][10], visited[10], v;

// Function to perform DFS
void DFS(int start)
{
    int i;

    printf("%d ", start);
    visited[start] = 1;

    for (i = 0; i < v; i++)
    {
        if (a[start][i] == 1 && visited[i] == 0)
        {
            DFS(i);
        }
    }
}

int main()
{
    int i, j, start;

    printf("Enter number of vertices: ");
    scanf("%d", &v);

    printf("Enter adjacency matrix:\n");

    for (i = 0; i < v; i++)
    {
        for (j = 0; j < v; j++)
        {
            scanf("%d", &a[i][j]);
        }
    }

    for (i = 0; i < v; i++)
    {
        visited[i] = 0;
    }

    printf("Enter starting vertex: ");
    scanf("%d", &start);

    printf("DFS Traversal: ");
    DFS(start);

    return 0;
}

Output

4 1 5 4
Enter starting vertex: 0
DFS Traversal: 0 2 3 1
