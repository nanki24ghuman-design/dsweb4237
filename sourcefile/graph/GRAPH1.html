Q1. Enter the vertices and edges and represent in adjacency matrix and print it.


#include <stdio.h>

int main()
{
  int a[10][10], v, e;
   int i, j, x, y;

  printf("Enter number of vertices: ");
  scanf("%d", &v);

  printf("Enter number of edges: ");
  scanf("%d", &e);

  // Initialize matrix with 0
  for (i = 0; i < v; i++)
   {
     for (j = 0; j < v; j++)
       {
        a[i][j] = 0;
      }
    }

  // Enter edges
  for (i = 0; i < e; i++)
   {
      printf("Enter two vertices: ");
      scanf("%d %d", &x, &y);

      a[x][y] = 1;
      a[y][x] = 1;
   }

  // Print adjacency matrix
  printf("\nAdjacency Matrix:\n");

   for (i = 0; i < v; i++)
  {
    for (j = 0; j < v; j++)
      {
        printf("%d ", a[i][j]);
      }
        printf("\n");
    }

  return 0;
}

Enter number of edges: 3

0 0 0 0

0 0 0 0
