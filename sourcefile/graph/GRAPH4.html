Q4. Traverse the adjacent list with depth first search


#include <stdio.h>
#include <stdlib.h>

struct node
{
    int vertex;
    struct node *next;
};

struct node *a[10];
int visited[10];

// Function to perform DFS
void DFS(int start)
{
    struct node *temp;

    printf("%d ", start);
    visited[start] = 1;

    temp = a[start];

    while (temp != NULL)
    {
        if (visited[temp->vertex] == 0)
        {
            DFS(temp->vertex);
        }

        temp = temp->next;
    }
}

int main()
{
    int v, e, i, x, y, start;
    struct node *temp;

    printf("Enter number of vertices: ");
    scanf("%d", &v);

    printf("Enter number of edges: ");
    scanf("%d", &e);

    for (i = 0; i < v; i++)
        a[i] = NULL;

    // Enter edges
    for (i = 0; i < e; i++)
    {
        printf("Enter two vertices: ");
        scanf("%d %d", &x, &y);

        temp = (struct node *)malloc(sizeof(struct node));
        temp->vertex = y;
        temp->next = a[x];
        a[x] = temp;

        temp = (struct node *)malloc(sizeof(struct node));
        temp->vertex = x;
        temp->next = a[y];
        a[y] = temp;
    }

    for (i = 0; i < v; i++)
        visited[i] = 0;

    printf("Enter starting vertex: ");
    scanf("%d", &start);

    printf("DFS Traversal: ");
    DFS(start);

    return 0;
}

Output

Enter number of vertices: 4
Enter number of edges: 4
Enter two vertices: 0 1
Enter two vertices: 0 2
Enter two vertices: 1 3
Enter two vertices: 2 3
Enter starting vertex: 0

DFS Traversal: 0 2 3 1
