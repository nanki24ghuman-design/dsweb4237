Q2. Create the graph as a adjacency list with operations such as insertion, deletion


#include <stdio.h>
#include <stdlib.h>

struct node
{
 int vertex;
 struct node *next;
};

struct node *a[10];

// Insert an edge
void insert(int x, int y)
{
  struct node *temp;

  temp = (struct node *)malloc(sizeof(struct node));
  temp->vertex = y;
  temp->next = a[x];
  a[x] = temp;

  temp = (struct node *)malloc(sizeof(struct node));
  temp->vertex = x;
  temp->next = a[y];
  a[y] = temp;
}

// Delete an edge
void deleteEdge(int x, int y)
{
  struct node *temp, *prev;

  prev = NULL;
  temp = a[x];

  while (temp != NULL)
  {
    if (temp->vertex == y)
   {
     if (prev == NULL)
      a[x] = temp->next;
     else
      prev->next = temp->next;
      free(temp);
     break;
   }

    prev = temp;
    temp = temp->next;
  }

    prev = NULL;
    temp = a[y];

  while (temp != NULL)
  {
   if (temp->vertex == x)
    {
      if (prev == NULL)
      a[y] = temp->next;
      else
      prev->next = temp->next;
      free(temp);
      break;
    }
    prev = temp;
    temp = temp->next;
  }
}

// Display the graph
void display(int v)
{
  int i;
  struct node *temp;
  for (i = 0; i < v; i++)
 {
    printf("%d -> ", i);
   temp = a[i];
   while (temp != NULL)
    {
      printf("%d -> ", temp->vertex);
      temp = temp->next;
    }

    printf("NULL\n");
  }
}

int main()
{
  int v, choice, x, y, i;
  printf("Enter number of vertices (maximum 10): ");
  scanf("%d", &v);

  if (v < 1 || v > 10)
  {
    printf("Invalid number of vertices.");
    return 0;
 }

  for (i = 0; i < v; i++)
    a[i] = NULL;

  do
  {
    printf("\n1. Insert Edge \n2. Delete Edge \n3. Display Graph \n4. Exit");
    printf("\nEnter your choice: ");
    scanf("%d", &choice);

    switch (choice)
    {
     case 1:
      printf("Enter two vertices: ");
      scanf("%d %d", &x, &y);
     if (x >= 0 && x < v && y >= 0 && y < v)
      insert(x, y);
     else
      printf("Invalid vertices.\n");
     break;

     case 2:
      printf("Enter edge to delete: ");
      scanf("%d %d", &x, &y);
     if (x >= 0 && x < v && y >= 0 && y < v)
      deleteEdge(x, y);
      else
        printf("Invalid vertices.\n");
     break;

     case 3:
      display(v);
      break;

     case 4:
       printf("Exiting program.\n");
        break;
     default:
      printf("Invalid choice.\n");
    }

    } while (choice != 4);

  return 0;
}
