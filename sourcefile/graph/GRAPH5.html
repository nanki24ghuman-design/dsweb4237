Q5. Traverse the adjacent matrix with breadth first search


#include <stdio.h>

int a[10][10], visited[10], queue[10];
int v;

void BFS(int start)
{
    int front = 0, rear = 0;
    int current, i;

    visited[start] = 1;
    queue[rear] = start;
    rear++;

    while (front < rear)
    {
        current = queue[front];
        front++;

        printf("%d ", current);

        for (i = 0; i < v; i++)
        {
            if (a[current][i] == 1 && visited[i] == 0)
            {
                visited[i] = 1;
                queue[rear] = i;
                rear++;
            }
        }
    }
}

int main()
{
    int i, j, start;

    printf("Enter number of vertices: ");
    scanf("%d", &v);

    printf("Enter adjacency matrix:\n");

    for (i = 0; i < v; i++)
    {
        for (j = 0; j < v; j++)
        {
            scanf("%d", &a[i][j]);
        }
    }

    for (i = 0; i < v; i++)
    {
        visited[i] = 0;
    }

    printf("Enter starting vertex: ");
    scanf("%d", &start);

    printf("BFS Traversal: ");
    BFS(start);

    return 0;
}
