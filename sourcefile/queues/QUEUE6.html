Q6. Implement the operations of circular queue in an string array (insert, delete, peek, isempty, isfull)


#include <stdio.h>
#include <string.h>

char q[5][20];
int front = -1, rear = -1;

void insert()
{
    char x[20];

    if ((rear + 1) % 5 == front)
        printf("Queue Full\n");
    else
    {
        printf("Enter string: ");
        scanf("%19s", x);

        if (front == -1)
            front = 0;

        rear = (rear + 1) % 5;
        strcpy(q[rear], x);
    }
}

void delete()
{
    if (front == -1)
        printf("Queue Empty\n");
    else
    {
        printf("Deleted = %s\n", q[front]);

        if (front == rear)
            front = rear = -1;
        else
            front = (front + 1) % 5;
    }
}

void peek()
{
    if (front == -1)
        printf("Queue Empty\n");
    else
        printf("Front = %s\n", q[front]);
}

void isempty()
{
    if (front == -1)
        printf("Queue is Empty\n");
    else
        printf("Queue is Not Empty\n");
}

void isfull()
{
    if ((rear + 1) % 5 == front)
        printf("Queue is Full\n");
    else
        printf("Queue is Not Full\n");
}

int main()
{
    int ch;

    while (1)
    {
        printf("\n1.Insert  2.Delete  3.Peek");
        printf("\n4.IsEmpty  5.IsFull  6.Exit");
        printf("\nEnter choice: ");
        scanf("%d", &ch);

        switch (ch)
        {
            case 1: insert(); break;
            case 2: delete(); break;
            case 3: peek(); break;
            case 4: isempty(); break;
            case 5: isfull(); break;
            case 6: return 0;
            default: printf("Invalid choice\n");
        }
    }
}
