Q1. Implement the operations of simple queue in an integer array (insert, delete, peek, isempty, isfull)


#include <stdio.h>
int q[5], front = -1, rear = -1;

void insert()
{
    int x;
    if (rear == 5 - 1)
        printf("Queue Full\n");
    else
    {
        printf("Enter value: ");
        scanf("%d", &x);
        if (front == -1)
            front = 0;
        rear++;
        q[rear] = x;
    }
}

void delete()
{
    if (front == -1)
        printf("Queue Empty\n");
    else
    {
        printf("Deleted = %d\n", q[front]);
        if (front == rear)
            front = rear = -1;
        else
            front++;
    }
}

void peek()
{
    if (front == -1)
        printf("Queue Empty\n");
    else
        printf("Front = %d\n", q[front]);
}

void isempty()
{
    if (front == -1)
        printf("Queue is Empty\n");
    else
        printf("Queue is Not Empty\n");
}

void isfull()
{
    if (rear == 5 - 1)
        printf("Queue is Full\n");
    else
        printf("Queue is Not Full\n");
}

int main()
{
    int ch;
    while (1)
    {
        printf("\n1.Insert 2.Delete 3.Peek");
        printf("\n4.IsEmpty 5.IsFull 6.Exit");
        printf("\nEnter choice: ");
        scanf("%d", &ch);

        switch (ch)
        {
            case 1: insert(); break;
            case 2: delete(); break;
            case 3: peek(); break;
            case 4: isempty(); break;
            case 5: isfull(); break;
            case 6: return 0;
            default: printf("Invalid choice");
        }
    }
    return 0;
}
