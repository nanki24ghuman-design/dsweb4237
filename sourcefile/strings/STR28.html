Q28. Create a dictionary of 10 words and their meanings with create, delete and search operations


#include <stdio.h>
#include <string.h>

struct dictionary
{
    char word[30];
    char meaning[100];
};

int main()
{
    struct dictionary d[20] = {
        {"Apple", "A fruit"},
        {"Book", "A set of pages"},
        {"Cat", "A domestic animal"},
        {"Dog", "A pet animal"},
        {"Earth", "Our planet"},
        {"Flower", "Part of a plant"},
        {"Garden", "A place for plants"},
        {"House", "A place to live"},
        {"Ink", "A liquid used for writing"},
        {"Juice", "A drink made from fruits"}
    };

    char word[30];
    int n = 10, i, j, choice, found;

    do
    {
        printf("\n1. Create\n2. Delete\n3. Search\n4. Exit\n");
        printf("Enter choice: ");
        scanf("%d", &choice);

        if(choice == 1)
        {
            if(n == 20)
            {
                printf("Dictionary is full");
                continue;
            }

            printf("Enter word: ");
            scanf("%29s", d[n].word);

            printf("Enter meaning: ");
            scanf(" %99[^\n]", d[n].meaning);

            n++;

            printf("Word created");
        }
        else if(choice == 2)
        {
            printf("Enter word to delete: ");
            scanf("%29s", word);

            found = 0;

            for(i = 0; i < n; i++)
            {
                if(strcmp(d[i].word, word) == 0)
                {
                    for(j = i; j < n - 1; j++)
                        d[j] = d[j + 1];

                    n--;
                    found = 1;

                    printf("Word deleted");
                    break;
                }
            }

            if(found == 0)
                printf("Word not found");
        }
        else if(choice == 3)
        {
            printf("Enter word to search: ");
            scanf("%29s", word);

            found = 0;

            for(i = 0; i < n; i++)
            {
                if(strcmp(d[i].word, word) == 0)
                {
                    printf("Meaning: %s", d[i].meaning);
                    found = 1;
                    break;
                }
            }

            if(found == 0)
                printf("Word not found");
        }
        else if(choice != 4)
            printf("Invalid choice");

    } while(choice != 4);

    return 0;
}

Output

1. Create
2. Delete
3. Search
4. Exit
Enter choice: 3
Enter word to search: Apple
Meaning: A fruit
