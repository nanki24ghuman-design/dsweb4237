Q16. Enter the string and a character, then print its index if it exists, else print -1


#include <stdio.h>

int main()
{
    char str[100], ch;
    int i, index = -1;

    printf("Enter a string: ");
    fgets(str, sizeof(str), stdin);

    printf("Enter a character: ");
    scanf("%c", &ch);

    for(i = 0; str[i] != '\0' && str[i] != '\n'; i++)
    {
        if(str[i] == ch)
        {
            index = i;
            break;
        }
    }

    printf("Index = %d", index);

    return 0;
}

Output

Enter a string: Pointer
Enter a character: t
Index = 4
