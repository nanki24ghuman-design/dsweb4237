Q13. Enter the string from user and copy its reverse into another string


#include <stdio.h>

int main()
{
    char str1[100], str2[100];
    int i = 0, j;

    printf("Enter a string: ");
    fgets(str1, sizeof(str1), stdin);

    while(str1[i] != '\0' && str1[i] != '\n')
        i++;

    for(j = 0; j < i; j++)
        str2[j] = str1[i - j - 1];

    str2[j] = '\0';

    printf("Reversed string: %s", str2);

    return 0;
}

Reversed string: retnioP
