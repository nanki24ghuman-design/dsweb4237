Q14. Enter the string and print the difference of ASCII values of adjacent characters


#include <stdio.h>

int main()
{
    char str[100];
    int i, len = 0, diff;

    printf("Enter a string: ");
    fgets(str, sizeof(str), stdin);

    while(str[len] != '\0' && str[len] != '\n')
        len++;

    for(i = 0; i < len - 1; i++)
    {
        diff = str[i] - str[i + 1];

        if(diff < 0)
            diff = -diff;

        printf("%d\n", diff);
    }

    return 0;
}

1
1
