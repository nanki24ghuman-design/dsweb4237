Q29. Enter 10 characters and convert them into their ASCII values in an integer array


#include <stdio.h>

int main()
{
    char str[11];
    int arr[10], i;

    printf("Enter 10 characters: ");
    scanf("%10s", str);

    for(i = 0; i < 10; i++)
        arr[i] = str[i];

    printf("ASCII values:\n");

    for(i = 0; i < 10; i++)
        printf("%d ", arr[i]);

    return 0;
}

Output

Enter 10 characters: ABCDEFGHIJ
ASCII values:
65 66 67 68 69 70 71 72 73 74
