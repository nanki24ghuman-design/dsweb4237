Q11. Enter the string from user and check whether it is a palindrome or not


#include <stdio.h>

int main()
{
    char str[100];
    int i = 0, j, flag = 1;

    printf("Enter a string: ");
    fgets(str, sizeof(str), stdin);

    while(str[i] != '\0' && str[i] != '\n')
        i++;

    j = i - 1;

    for(i = 0; i < j; i++, j--)
    {
        if(str[i] != str[j])
        {
            flag = 0;
            break;
        }
    }

    if(flag == 1)
        printf("Palindrome");
    else
        printf("Not a palindrome");

    return 0;
}

Output

Enter a string: level
Palindrome
