Q6. Enter the string from user and find the length of the string


#include <stdio.h>

int main()
{
    char str[100];
    int i = 0;

    printf("Enter a string: ");
    fgets(str, sizeof(str), stdin);

    while(str[i] != '\0' && str[i] != '\n')
        i++;

    printf("Length = %d", i);

    return 0;
}

Output

Enter a string: Hello
Length = 5
