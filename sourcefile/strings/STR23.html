Q23. Enter a sentence and a word, then count the occurrences of that word


#include <stdio.h>
#include <string.h>

int main()
{
    char str[300], word[50];
    int i = 0, j, k, count = 0, match;

    printf("Enter a sentence: ");
    fgets(str, sizeof(str), stdin);

    printf("Enter a word: ");
    scanf("%49s", word);

    while(str[i] != '\0' && str[i] != '\n')
    {
        while(str[i] == ' ')
            i++;

        j = i;

        while(str[i] != ' ' && str[i] != '\0' && str[i] != '\n')
            i++;

        if(i > j)
        {
            match = 1;

            if(i - j != strlen(word))
                match = 0;
            else
            {
                for(k = 0; word[k] != '\0'; k++)
                {
                    if(str[j + k] != word[k])
                    {
                        match = 0;
                        break;
                    }
                }
            }

            if(match == 1)
                count++;
        }
    }

    printf("Occurrences = %d", count);

    return 0;
}

Enter a sentence: red blue red green red
Enter a word: red
Occurrences = 3
