Q22. Enter the string from user, sort it in ascending order and print it


#include <stdio.h>

int main()
{
    char str[100], temp;
    int i, j, len = 0;

    printf("Enter a string: ");
    fgets(str, sizeof(str), stdin);

    while(str[len] != '\0' && str[len] != '\n')
        len++;

    for(i = 0; i < len - 1; i++)
    {
        for(j = 0; j < len - i - 1; j++)
        {
            if(str[j] > str[j + 1])
            {
                temp = str[j];
                str[j] = str[j + 1];
                str[j + 1] = temp;
            }
        }
    }

    str[len] = '\0';

    printf("Sorted string: %s", str);

    return 0;
}

Output

Enter a string: zyxw
Sorted string: wxyz
