Q20. Enter the string and count characters, words, spaces and lines


#include <stdio.h>

int main()
{
    char str[1000];
    int i, ch, len = 0, words = 0;
    int spaces = 0, lines = 0, inword = 0;

    printf("Enter text (# at start of a line to stop):\n");

    while((ch = getchar()) != EOF && len < 999)
    {
        if(ch == '#' && (len == 0 || str[len - 1] == '\n'))
            break;

        str[len++] = ch;
    }

    str[len] = '\0';

    for(i = 0; i < len; i++)
    {
        if(str[i] == ' ')
            spaces++;

        if(str[i] == '\n')
            lines++;

        if(str[i] != ' ' && str[i] != '\n' && str[i] != '\t')
        {
            if(inword == 0)
                words++;

            inword = 1;
        }
        else
            inword = 0;
    }

    if(len > 0 && str[len - 1] != '\n')
        lines++;

    printf("Characters = %d\n", len);
    printf("Words = %d\n", words);
    printf("Spaces = %d\n", spaces);
    printf("Lines = %d\n", lines);

    return 0;
}

Output

Hello World

Lines = 2
