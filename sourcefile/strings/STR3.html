Q3. Enter the string from user and convert into sentence case


#include <stdio.h>

int main()
{
    char str[100];
    int i, first = 1;

    printf("Enter a string: ");
    fgets(str, sizeof(str), stdin);

    for(i = 0; str[i] != '\0'; i++)
    {
        if(str[i] >= 'A' && str[i] <= 'Z')
            str[i] = str[i] + 32;

        if(first == 1 && str[i] >= 'a' && str[i] <= 'z')
        {
            str[i] = str[i] - 32;
            first = 0;
        }
    }

    printf("Sentence case: %s", str);

    return 0;
}

Output

Enter a string: dATA sTRUCTURE
Sentence case: Data structure
