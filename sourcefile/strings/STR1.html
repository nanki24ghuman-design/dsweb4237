Q1. Enter the string from user and convert into uppercase


#include <stdio.h>

int main()
{
    char str[100];
    int i;

    printf("Enter a string: ");
    fgets(str, sizeof(str), stdin);

    for(i = 0; str[i] != '\0'; i++)
    {
        if(str[i] >= 'a' && str[i] <= 'z')
            str[i] = str[i] - 32;
    }

    printf("Uppercase: %s", str);

    return 0;
}

Output

Enter a string: Data Structure
Uppercase: DATA STRUCTURE
