Q21. Print all the permutations of the string "ABCDE"


#include <stdio.h>
#include <string.h>

void swap(char *a, char *b)
{
    char temp = *a;
    *a = *b;
    *b = temp;
}

void permute(char str[], int start, int end)
{
    int i;

    if(start == end)
        printf("%s\n", str);
    else
    {
        for(i = start; i <= end; i++)
        {
            swap(&str[start], &str[i]);
            permute(str, start + 1, end);
            swap(&str[start], &str[i]);
        }
    }
}

int main()
{
    char str[] = "ABCDE";

    permute(str, 0, strlen(str) - 1);

    return 0;
}
