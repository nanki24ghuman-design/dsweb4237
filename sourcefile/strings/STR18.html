Q18. Enter the string from user and convert it into an integer value


#include <stdio.h>

int main()
{
    char str[100];
    int i = 0, num = 0, sign = 1;

    printf("Enter a number: ");
    scanf("%99s", str);

    if(str[0] == '-')
    {
        sign = -1;
        i = 1;
    }

    for(; str[i] != '\0'; i++)
        num = num * 10 + (str[i] - '0');

    num = num * sign;

    printf("Integer value = %d", num);

    return 0;
}

Output

Enter a number: 127
Integer value = 127
