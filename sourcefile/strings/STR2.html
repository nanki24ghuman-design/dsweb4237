Q2. Enter the string from user and convert into lowercase


#include <stdio.h>

int main()
{
    char str[100];
    int i;

    printf("Enter a string: ");
    fgets(str, sizeof(str), stdin);

    for(i = 0; str[i] != '\0'; i++)
    {
        if(str[i] >= 'A' && str[i] <= 'Z')
            str[i] = str[i] + 32;
    }

    printf("Lowercase: %s", str);

    return 0;
}

Output

Enter a string: DATA STRUCTURE
Lowercase: data structure
