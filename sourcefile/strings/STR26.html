Q26. Enter the string, print the first and second words, and find the start index of the second word


#include <stdio.h>

int main()
{
    char str[100];
    int i = 0, start, end;

    printf("Enter a string: ");
    fgets(str, sizeof(str), stdin);

    while(str[i] == ' ')
        i++;

    start = i;

    while(str[i] != ' ' && str[i] != '\0' && str[i] != '\n')
        i++;

    end = i;

    printf("First word: ");

    for(i = start; i < end; i++)
        printf("%c", str[i]);

    i = end;

    while(str[i] == ' ')
        i++;

    if(str[i] == '\0' || str[i] == '\n')
    {
        printf("\nSecond word not found");
        return 0;
    }

    start = i;

    printf("\nSecond word: ");

    while(str[i] != ' ' && str[i] != '\0' && str[i] != '\n')
    {
        printf("%c", str[i]);
        i++;
    }

    printf("\nStart index of second word = %d", start);

    return 0;
}

Output

Enter a string: Data Structure
First word: Data
Second word: Structure
Start index of second word = 5
