Q19. Enter the string from user and convert it into a float value


#include <stdio.h>

int main()
{
    char str[100];
    int i = 0, sign = 1;
    float num = 0, place = 0.1;

    printf("Enter a number: ");
    scanf("%99s", str);

    if(str[0] == '-')
    {
        sign = -1;
        i = 1;
    }

    while(str[i] >= '0' && str[i] <= '9')
    {
        num = num * 10 + (str[i] - '0');
        i++;
    }

    if(str[i] == '.')
    {
        i++;

        while(str[i] >= '0' && str[i] <= '9')
        {
            num = num + (str[i] - '0') * place;
            place = place / 10;
            i++;
        }
    }

    num = num * sign;

    printf("Float value = %.2f", num);

    return 0;
}

Output

Enter a number: 12.5
Float value = 12.50
