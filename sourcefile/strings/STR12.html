Q12. Enter the string from user and print each character in new line


#include <stdio.h>

int main()
{
    char str[100];
    int i;

    printf("Enter a string: ");
    fgets(str, sizeof(str), stdin);

    for(i = 0; str[i] != '\0' && str[i] != '\n'; i++)
        printf("%c\n", str[i]);

    return 0;
}

Output

Enter a string: DOG
D
O
G
