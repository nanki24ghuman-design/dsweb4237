Q27. Display a string with user-selected alignment: 1 left, 2 center, 3 right


#include <stdio.h>

int main()
{
    char str[100];
    int i, len = 0, choice, spaces;

    printf("Enter a string: ");
    fgets(str, sizeof(str), stdin);

    while(str[len] != '\0' && str[len] != '\n')
        len++;

    printf("1. Left\n2. Center\n3. Right\n");
    printf("Enter choice: ");
    scanf("%d", &choice);

    if(len > 40)
    {
        printf("String is too long");
        return 0;
    }

    if(choice == 1)
        spaces = 0;
    else if(choice == 2)
        spaces = (40 - len) / 2;
    else if(choice == 3)
        spaces = 40 - len;
    else
    {
        printf("Invalid choice");
        return 0;
    }

    for(i = 0; i < spaces; i++)
        printf(" ");

    printf("%s", str);

    return 0;
}
