Q8. Enter two strings and concatenate them into another string


#include <stdio.h>

int main()
{
    char str1[100], str2[50], str3[150];
    int i = 0, j = 0;

    printf("Enter first string: ");
    fgets(str1, sizeof(str1), stdin);

    printf("Enter second string: ");
    fgets(str2, sizeof(str2), stdin);

    while(str1[i] != '\0' && str1[i] != '\n')
    {
        str3[i] = str1[i];
        i++;
    }

    while(str2[j] != '\0' && str2[j] != '\n')
    {
        str3[i] = str2[j];
        i++;
        j++;
    }

    str3[i] = '\0';

    printf("Concatenated string: %s", str3);

    return 0;
}

Enter second string: Structure
Concatenated string: DataStructure
