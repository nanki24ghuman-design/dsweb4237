Q10. Enter the string, enter the index and print the character at that index


#include <stdio.h>

int main()
{
    char str[100];
    int i = 0, index;

    printf("Enter a string: ");
    fgets(str, sizeof(str), stdin);

    while(str[i] != '\0' && str[i] != '\n')
        i++;

    printf("Enter index number: ");
    scanf("%d", &index);

    if(index >= 0 && index < i)
        printf("Character = %c", str[index]);
    else
        printf("Invalid index");

    return 0;
}

Output

Enter a string: Pointer
Enter index number: 2
Character = i
