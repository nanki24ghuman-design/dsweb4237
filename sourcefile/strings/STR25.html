Q25. Enter the string and eliminate spaces from both ends


#include <stdio.h>

int main()
{
    char str[100];
    int start = 0, end = 0, i;

    printf("Enter a string: ");
    fgets(str, sizeof(str), stdin);

    while(str[end] != '\0' && str[end] != '\n')
        end++;

    end--;

    while(str[start] == ' ')
        start++;

    while(end >= start && str[end] == ' ')
        end--;

    printf("Trimmed string: ");

    for(i = start; i <= end; i++)
        printf("%c", str[i]);

    return 0;
}
