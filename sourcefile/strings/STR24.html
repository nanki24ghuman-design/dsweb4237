Q24. Enter a sentence and replace a word with another word


#include <stdio.h>
#include <string.h>

int main()
{
    char str[300], word[50], rep[50], result[500];
    int i = 0, j, k, r = 0, match, first = 1;

    printf("Enter a sentence: ");
    fgets(str, sizeof(str), stdin);

    printf("Enter word to replace: ");
    scanf("%49s", word);

    printf("Enter new word: ");
    scanf("%49s", rep);

    while(str[i] != '\0' && str[i] != '\n')
    {
        while(str[i] == ' ')
        {
            if(r > 0 && result[r - 1] != ' ')
                result[r++] = ' ';
            i++;
        }

        if(str[i] == '\0' || str[i] == '\n')
            break;

        j = i;

        while(str[i] != ' ' && str[i] != '\0' && str[i] != '\n')
            i++;

        match = 1;

        if(i - j != strlen(word))
            match = 0;
        else
        {
            for(k = 0; word[k] != '\0'; k++)
            {
                if(str[j + k] != word[k])
                {
                    match = 0;
                    break;
                }
            }
        }

        if(!first && r > 0 && result[r - 1] != ' ')
            result[r++] = ' ';

        if(match == 1)
        {
            for(k = 0; rep[k] != '\0'; k++)
                result[r++] = rep[k];
        }
        else
        {
            for(k = j; k < i; k++)
                result[r++] = str[k];
        }

        first = 0;
    }

    result[r] = '\0';

    printf("New sentence: %s", result);

    return 0;
}

Output

Enter a sentence: We like C and we like Java
Enter word to replace: like
Enter new word: love
New sentence: We love C and we love Java
