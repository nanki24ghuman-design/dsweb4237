Q2. Enter an array from user and print the array in reverse order


#include<stdio.h>

int main()
{
  int i, n;
  printf("Enter the size of array : ");
  scanf("%d", &n);

  int ar[n];
  printf("Enter the values of arrays ");

  for(i = 0; i < n; i++)
  {
    scanf("%d", &ar[i]);
  }

  printf("The reversed order elements of array are: ");
  for(i = n - 1; i >= 0; i--)
  {
    printf("%d ", ar[i]);
  }

 return 0;
}

Output

Enter the size of array : 5
Enter the values of arrays: 1 2 3 4 5

The reversed order elements of array are: 5 4 3 2 1
