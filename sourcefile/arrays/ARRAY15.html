Q15. Enter an array from user and print all the palindrome numbers


#include<stdio.h>
int main()
{
    int arr[5],i,n,rev,rem;
    printf("Enter any five numbers: ");
    for(i=0;i<5;i++)
    {
        scanf("%d",&arr[i]);
    }
    printf("Palindrome numbers are: ");
    for(i=0;i<5;i++)
    {
        n=arr[i];
        rev=0;
        while(n>0)
        {
            rem=n%10;
            rev=rev*10+rem;
            n=n/10;
        }
        if(arr[i]==rev)
        {
            printf("%d ",arr[i]);
        }
    }
    return 0;
}

Output

Enter any five numbers: 1001
234
4545
6556
6789
Palindrome numbers are: 1001 6556
