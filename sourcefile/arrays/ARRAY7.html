Q7. Enter an array from user and square all the values of array


#include<stdio.h>
int main()
{
    int arr[5];
    printf("Enter any five numbers ");
    for(int i=0;i<5;i++)
    {
        scanf("%d",&arr[i]);
    }
     
    printf("Squares of all the numbers are : ");
  
    for(int i=0;i<5;i++){
            printf("%d \n",arr[i]*arr[i]);
        }
     return 0;
}

Output

Enter any five numbers 9  8  7  7  6
Squares of all the numbers are : 81  64  49  49  36
