Q10. Enter an array from user and add all values


//Enter an array from user and add all values ?
#include<stdio.h>
int main()
{
    int arr[5],i,sum=0;
    printf("Enter any five numbers: ");
    for(i=0;i<5;i++)
    {
        scanf("%d",&arr[i]);
    }
    for(i=0;i<5;i++)
    {
        sum=sum+arr[i];
    }
    printf("Sum of all values = %d",sum);
    return 0;
}

Output

Enter any five numbers: 10
79
7
8
7
Sum of all values = 111
