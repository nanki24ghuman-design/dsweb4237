Q3. Enter an array from user and copy the array into second array


#include<stdio.h>
int main()
{
    int ar1[15], ar2[15],n, i;
    printf("Enter the size of array : ");
    scanf("%d",&n);
    printf("Enter the elements of first array:\n");
    for(i=0; i<n; i++)
    {
        scanf("%d",&ar1[i]);
    }
    for(i=0; i<n; i++)
    {
        ar2[i]=ar1[i];
    }
    printf("The elements of second array are:\n");
    for(i=0; i<n; i++)
    {
        printf("%d ",ar2[i]);
    }
    return 0;
}

Output

Enter the size of array :4
Enter the elements of first array: 2 5 6 7 
The elements of second array are:  2 5 6 7
