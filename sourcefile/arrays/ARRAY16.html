Q16. Enter two arrays from the user and check they are same or not


#include<stdio.h>
int main()
{
    int a[5],b[5],i,same=1;
    printf("Enter five elements of first array: ");
    for(i=0;i<5;i++)
    {
        scanf("%d",&a[i]);
    }
    printf("Enter five elements of second array: ");
    for(i=0;i<5;i++)
    {
        scanf("%d",&b[i]);
    }
    for(i=0;i<5;i++)
    {
        if(a[i]!=b[i])
        {
            same=0;
            break;
        }
    }
    if(same==1)
    {
        printf("Both arrays are same");
    }
    else
    {
        printf("Both arrays are not same");
    }
    return 0;
}

Output

Enter five elements of first array: 1 2 3 3 5
Enter five elements of second array: 1 2 3 4 7
Both arrays are not same
