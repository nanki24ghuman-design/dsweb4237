Q13. Enter an array from user and count the duplicate values


#include<stdio.h>
int main()
{
    int arr[5],i,j,count=0;
    printf("Enter any five numbers: ");
    for(i=0;i<5;i++)
    {
        scanf("%d",&arr[i]);
    }
    for(i=0;i<5;i++)
    {
        for(j=i+1;j<5;j++)
        {
            if(arr[i]==arr[j])
            {
                count++;
                break;
            }
        }
    }
    printf("Total duplicate values = %d",count);
    return 0;
}

Output

Enter any five numbers: 6
5
4
3
4
Total duplicate values = 1
