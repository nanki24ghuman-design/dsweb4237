Q8. Enter an array from user and find maximum and minimum value


#include<stdio.h>

int main()
{
    int arr[5], i, min, max;
    printf("Enter any five numbers: ");
    for(i=0; i<5; i++)
    {
        scanf("%d", &arr[i]);
    }
    min = arr[0];
    max = arr[0];
    for(i=1; i<5; i++)
    {
        if(arr[i] < min)
        {
            min = arr[i];
        }

        if(arr[i] > max)
        {
            max = arr[i];
        }
    }
    printf("Minimum number = %d\n", min);
    printf("Maximum number = %d", max);
    return 0;
}

Output

Enter any five numbers: 9
8
7
10
7
Minimum number = 7
Maximum number = 10
