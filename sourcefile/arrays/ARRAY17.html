Q17. Merge two sorted arrays into one array in sorted order


#include<stdio.h>
int main()
{
    int a[5],b[5],c[10],i,j,k;
    printf("Enter five elements of first sorted array: ");
    for(i=0;i<5;i++)
    {
        scanf("%d",&a[i]);
    }
    printf("Enter five elements of second sorted array: ");
    for(i=0;i<5;i++)
    {
        scanf("%d",&b[i]);
    }
    for(i=0;i<5;i++)
    {
        c[i]=a[i];
    }
    for(i=0;i<5;i++)
    {
        c[i+5]=b[i];
    }
    for(i=0;i<10;i++)
    {
        for(j=i+1;j<10;j++)
        {
            if(c[i]>c[j])
            {
                k=c[i];
                c[i]=c[j];
                c[j]=k;
            }
        }
    }
    printf("Merged sorted array: ");
    for(i=0;i<10;i++)
    {
        printf("%d ",c[i]);
    }
    return 0;
}

Output

Enter five elements of first sorted array: 7 8 9 10 11 12
Enter five elements of second sorted array: 6 8 9 11 12
Merged sorted array: 6 7 8 8 9 9 10 11 11 12
