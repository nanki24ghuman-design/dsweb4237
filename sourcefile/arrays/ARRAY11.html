Q11. Enter an array from user and find the duplicate values


#include<stdio.h>
int main()
{
    int arr[5],i,j;
    printf("Enter any five numbers: ");
    for(i=0;i<5;i++)
    {
        scanf("%d",&arr[i]);
    }
    printf("Duplicate values are: ");
    for(i=0;i<5;i++)
    {
        for(j=i+1;j<5;j++)
        {
            if(arr[i]==arr[j])
            {
                printf("%d ",arr[i]);
                break;
            }
        }
    }
    return 0;
}

Output

Enter any five numbers: 7
7
6
6
8
Duplicate values are: 7 6
