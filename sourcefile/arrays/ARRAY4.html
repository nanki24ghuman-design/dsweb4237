Q4. Enter an array from user and copy it into another array in reverse order


#include<stdio.h>
int main()
{
    int ar[50], b[50], i, n;
    printf("Enter the size of array: ");
    scanf("%d", &n);
    printf("Enter the elements of array:\n");
    for(i = 0; i < n; i++)
    {
        scanf("%d", &ar[i]);
    }
    for(i = 0; i < n; i++)
    {
        b[i] = ar[n - 1 - i];
    }
    printf("Original array: ");
    for(i = 0; i < n; i++)
    {
        printf("%d ", ar[i]);
    }
    printf("\nCopied array in reverse order: ");
    for(i = 0; i < n; i++)
    {
        printf("%d ", b[i]);
    }
    return 0;
}

Output

Enter the size of array: 5
Enter the elements of array:10 20 30 40 50
Original array: 10 20 30 40 50
Copied array in reverse order: 50 40 30 20 10
