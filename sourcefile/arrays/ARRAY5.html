Q5. Enter an array from user and print the number which is even


#include <stdio.h>
int main()
{
  int ar[50], i, n;
  printf("Enter the size of array: ");
  scanf("%d", &n);
  printf("Enter the elements of array:\n");

  for(i = 0; i < n; i++)
  {
      scanf("%d", &ar[i]);
  }
  printf("Even numbers in the array are: ");
  for(i = 0; i < n; i++)
  {
    if(ar[i] % 2 == 0)
    {
        printf("%d ", ar[i]);
    }
  }

  return 0;
}

Output

Enter the size of array: 5
Enter any five numbers 6 7 8 9 10
Even numbers are : 6 8 10
