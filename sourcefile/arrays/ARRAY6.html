Q6. Enter an array from user and print the number which is positive


#include<stdio.h>
int main()
{
  int arr[5];
  printf("Enter any five numbers ");
  for(int i=0;i<=4;i++)
  {
      scanf("%d",&arr[i]);
  }
   
  printf("Positive  numbers are : ");

  for(int i=0;i<=4;i++){
      if(arr[i]>0){
          printf("%d ",arr[i]);
      }}
   return 0;
}

Output

Enter any five numbers 9  7 -4  -3  13
Positive  numbers are : 9  7  13
