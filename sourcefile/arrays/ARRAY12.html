Q12. Enter an array from user and remove the duplicate values and fill with zero


#include<stdio.h>
int main()
{
    int arr[5],i,j;
    printf("Enter any five numbers: ");
    for(i=0;i<5;i++)
    {
        scanf("%d",&arr[i]);
    }
    for(i=0;i<5;i++)
    {
        for(j=i+1;j<5;j++)
        {
            if(arr[i]==arr[j])
            {
                arr[j]=0;
            }
        }
    }
    printf("Array after removing duplicates: ");
    for(i=0;i<5;i++)
    {
        printf("%d ",arr[i]);
    }
    return 0;
}

Output

Enter any five numbers: 10
11
8
7
8
Array after removing duplicates: 10 11 8 7 0
