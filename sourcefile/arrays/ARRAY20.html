Q20. Trapping Rain Water Problem: Given n non-negative integers representing an elevation map where the width of each bar is 1, compute how much water it can trap after raining.


#include<stdio.h>
int main()
{
    int arr[5],i,j,left,right,water=0;
    printf("Enter five heights: ");
    for(i=0;i<5;i++)
    {
        scanf("%d",&arr[i]);
    }
    for(i=1;i<4;i++)
    {
        left=arr[0];
        right=arr[4];
        for(j=0;j<i;j++)
        {
            if(arr[j]>left)
            {
                left=arr[j];
            }
        }
        for(j=i+1;j<5;j++)
        {
            if(arr[j]>right)
            {
                right=arr[j];
            }
        }
        if(left<right)
        {
            water=water+left-arr[i];
        }
        else
        {
            water=water+right-arr[i];
        }
    }
    printf("Trapped water = %d",water);
    return 0;
}

Output

Enter five heights: 5 6 7 0 8
Trapped water = 5
