Q18. Find the local minima in the array? (An element in an array is a local minima if it less than the element after it, and the element before it)


#include<stdio.h>
int main()
{
    int arr[5],i;
    printf("Enter any five numbers: ");
    for(i=0;i<5;i++)
    {
        scanf("%d",&arr[i]);
    }
    printf("Local minima are: ");
    for(i=1;i<4;i++)
    {
        if(arr[i]<arr[i-1] && arr[i]<arr[i+1])
        {
            printf("%d ",arr[i]);
        }
    }
    return 0;
}

Output

Enter any five numbers: 9 8 7 11 10 8
Local minima are: 7
