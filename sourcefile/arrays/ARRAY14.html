Q14. Enter an array from user and print all possible combinations of the elements


#include<stdio.h>
int main()
{
    int arr[3],i,j;
    printf("Enter any three numbers: ");
    for(i=0;i<3;i++)
    {
        scanf("%d",&arr[i]);
    }
    printf("All possible combinations are:\n");
    for(i=0;i<3;i++)
    {
        for(j=0;j<3;j++)
        {
            if(i!=j)
            {
                printf("%d %d\n",arr[i],arr[j]);
            }
        }
    }
    return 0;
}

Output

Enter any three numbers: 10 11 13
All possible combinations are:
10 11
10 13
11 10
11 13
13 10
13 11
