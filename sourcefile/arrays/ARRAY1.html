Q1. Enter an array from user and display all the value


#include<stdio.h>
int main()
{
    int ar[50],i,n;
    printf("Enter the size of array : ");
    scanf("%d",&n);
    printf("Enter the values of arrays ");
        for(i=0;i<n;i++)
        {
        scanf("%d",&ar[i]);
        }
    printf("The elements of array are : ");
    for(i=0;i<n;i++)
    {
        printf("%d ",ar[i]);
    }
return 0;

}

Output

Enter the size of array : 4
Enter the values of arrays 5
4
5
5
The elements of array are : 5 4 5 5
