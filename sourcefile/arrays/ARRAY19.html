Q19. Find the local maxima in the array


#include<stdio.h>
int main()
{
    int arr[5],i;
    printf("Enter any five numbers: ");
    for(i=0;i<5;i++)
    {
        scanf("%d",&arr[i]);
    }
    printf("Local maxima are: ");
    for(i=1;i<4;i++)
    {
        if(arr[i]>arr[i-1] && arr[i]>arr[i+1])
        {
            printf("%d ",arr[i]);
        }
    }
    return 0;
}

Output

Enter any five numbers: 10 11 12 13 10 9
Local maxima are: 13
