Q9.Enter an array from user and find the k-th smallest value


#include<stdio.h>

int main()
{
    int arr[5], i, j, temp, k;
    printf("Enter any five numbers: ");
    for(i=0; i<5; i++)
    {
        scanf("%d", &arr[i]);
    }
    printf("Enter the value of k: ");
    scanf("%d", &k);
    for(i=0; i<5; i++)
    {
        for(j=i+1; j<5; j++)
        {
            if(arr[i] > arr[j])
            {
                temp = arr[i];
                arr[i] = arr[j];
                arr[j] = temp;
            }
        }
    }
    if(k>=1 && k<=5)
    {
        printf("K-th smallest value = %d", arr[k-1]);
    }
    else
    {
        printf("Invalid value of k");
    }

    return 0;
}

Output

Enter any five numbers: 9  8  10  7  11
Enter the value of k: 5
K-th smallest value = 11
