Q14. Print preorder traversal to postorder traversal of binary search tree


#include <stdio.h>

void printPostorder(int pre[], int n) {
    if (n <= 0)
        return;

    int i = 0;

    // Find elements belonging to left subtree
    while (i < n && pre[i] < pre[0])
        i++;

    // Print left subtree
    printPostorder(pre + 1, i);

    // Print right subtree
    printPostorder(pre + i + 1, n - i - 1);

    // Print root
    printf("%d ", pre[0]);
}

int main() {
    int n, pre[100];

    printf("Enter number of nodes: ");
    scanf("%d", &n);

    if (n < 1 || n > 100) {
        printf("Invalid number of nodes.\n");
        return 0;
    }

    printf("Enter preorder traversal: ");
    for (int i = 0; i < n; i++)
        scanf("%d", &pre[i]);

    printf("Postorder traversal: ");
    printPostorder(pre, n);

    return 0;
}

Output

Enter number of nodes: 6
Enter preorder traversal: 14 24 29 39 44 64
Postorder traversal: 64 44 39 29 24 14
