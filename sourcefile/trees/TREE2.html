Q2. Create binary search tree dynamically (insert, delete, search, preorder, inorder, postorder, minimum element)


#include<stdio.h>
#include<stdlib.h>
struct BST{
    int data;
    struct BST *left,*right;
};
struct BST *start=NULL, *temp ,*search;
void insert(){
    int value;
temp=(struct BST*)malloc(sizeof(struct BST));
printf("enter the value you want to insert:\n");
scanf("%d",&value);
if(temp!=NULL){
    temp->data=value;
    temp->left=NULL;
    temp->right=NULL;
}
if(start==NULL){
    start=temp;
}
else{
    search=start;
    while(search!=NULL){
        if(temp->data==search->data){
            free(temp);
            break;
        }
        else if(temp->data<search->data){
            if(search->left!=NULL){
                search=search->left;
            }
            else{
                search->left=temp;
                break;
            }
            
        }
        else if(temp->data>search->data){
                if(search->right!=NULL){
                    search=search->right;
                }
                else{
                    search->right=temp;
                    break;
                }
            }
    }
}
printf("insertion completed\n");
}



void preorder(struct BST*temp){
    if(temp==NULL){
        return;
    }
    printf("%d",temp->data);
    preorder(temp->left);
    preorder(temp->right);
}
void inorder(struct BST*temp){
    if(temp!=NULL){
        inorder(temp->left);
        printf("%d",temp->data);
        inorder(temp->right);
    }
}
void postorder(struct BST*temp){
    if(temp==NULL){
        return;
    }
    postorder(temp->left);
    postorder(temp->right);
    printf("%d",temp->data);
}
void minimum(){
    if(start==NULL){
        printf("tree is empty\n");
        return;
    }
    search=start;
    while(search->left!=NULL){
        search=search->left;
    }
    printf("minimum elemnet =%d\n",search->data);
}
struct BST*deletenode(struct BST *temp,int value){
    struct BST*child;
    if(temp==NULL)
        return NULL;
    if(value<temp->data){
        temp->left =deletenode(temp->left,value);
    }
    else if(value>temp->data){
        temp->right=deletenode(temp->right,value);
    }
    else{
        if(temp->left==NULL&&temp->right==NULL){
            free(temp);
            return NULL;
        }
        if(temp->left==NULL){
            child=temp->right;
            free(temp);
            return child;
        }
        if(temp->right==NULL){
            child=temp->left;
            free(temp);
            return child;
        }
        child=temp->right;
        while(child->left!=NULL){
            child=child->left;
        }
        temp->data=child->data;
        temp->right=deletenode(temp->right,child->data);
    }
    return temp;
}
void deletion(){
    int value;
    if(start==NULL){
        printf("tree is empty\n");
        return;
    }
    printf("enter the value you want to delete:");
    scanf("%d",&value);
    start=deletenode(start,value);
    printf("deletion completed \n");
}

int choices;
 do{

    switch(choices){

     case 5:
