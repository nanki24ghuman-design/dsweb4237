Q13. Find kth smallest element in a binary search tree through array


#include <stdio.h>
#include <stdlib.h>

struct Node {
    int data;
    struct Node *left, *right;
};

struct Node* createNode(int data) {
    struct Node* newNode =
        (struct Node*)malloc(sizeof(struct Node));

    newNode->data = data;
    newNode->left = NULL;
    newNode->right = NULL;

    return newNode;
}

struct Node* insert(struct Node* root, int data) {
    if (root == NULL)
        return createNode(data);

    if (data < root->data)
        root->left = insert(root->left, data);
    else if (data > root->data)
        root->right = insert(root->right, data);

    return root;
}

// Store elements in an array using inorder traversal
void inorder(struct Node* root, int arr[], int *i) {
    if (root == NULL)
        return;

    inorder(root->left, arr, i);
    arr[(*i)++] = root->data;
    inorder(root->right, arr, i);
}

int main() {
    struct Node* root = NULL;
    int arr[100], n, value, k, i = 0;

    printf("Enter number of nodes: ");
    scanf("%d", &n);

    if (n < 1 || n > 100) {
        printf("Invalid number of nodes.\n");
        return 0;
    }

    printf("Enter node values: ");
    for (int j = 0; j < n; j++) {
        scanf("%d", &value);
        root = insert(root, value);
    }

    printf("Enter K: ");
    scanf("%d", &k);

    inorder(root, arr, &i);

    if (k < 1 || k > i) {
        printf("Invalid K value.\n");
    } else {
        printf("Kth smallest element = %d\n",
               arr[k - 1]);
    }

    return 0;
}
