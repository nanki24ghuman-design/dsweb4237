Q8. Implement the operations of doubly circular linked list storing string values (insert at begin,insert at pos,insert at end,delete at begin,delete at end, delete at pos, isempty, isfull, display, search)


#include <stdio.h>
#include <stdlib.h>
#include <string.h>

struct node
{
    char data[100];
    struct node *prev, *next;
};

struct node *start = 0;

struct node* create()
{
    struct node *temp;
    temp = (struct node*)malloc(sizeof(struct node));

    if(temp == 0)
    {
        printf("Memory Full\n");
        return 0;
    }

    printf("Enter string: ");
    scanf("%99s", temp->data);

    return temp;
}

int isempty()
{
    return start == 0;
}

int isfull()
{
    struct node *temp = (struct node*)malloc(sizeof(struct node));

    if(temp == 0)
        return 1;

    free(temp);
    return 0;
}

void insert_begin()
{
    struct node *temp = create();
    struct node *last;

    if(temp == 0)
        return;

    if(start == 0)
    {
        start = temp;
        temp->next = temp;
        temp->prev = temp;
        return;
    }

    last = start->prev;

    temp->next = start;
    temp->prev = last;
    last->next = temp;
    start->prev = temp;
    start = temp;
}

void insert_end()
{
    struct node *temp = create();
    struct node *last;

    if(temp == 0)
        return;

    if(start == 0)
    {
        start = temp;
        temp->next = temp;
        temp->prev = temp;
        return;
    }

    last = start->prev;

    temp->next = start;
    temp->prev = last;
    last->next = temp;
    start->prev = temp;
}

void insert_pos()
{
    int pos, i;
    struct node *temp, *p;

    printf("Enter position: ");
    scanf("%d", &pos);

    if(pos < 1)
    {
        printf("Invalid position\n");
        return;
    }

    if(pos == 1)
    {
        insert_begin();
        return;
    }

    if(start == 0)
    {
        printf("Invalid position\n");
        return;
    }

    p = start;

    for(i = 1; i < pos - 1 && p->next != start; i++)
        p = p->next;

    if(i != pos - 1)
    {
        printf("Invalid position\n");
        return;
    }

    temp = create();

    if(temp == 0)
        return;

    temp->next = p->next;
    temp->prev = p;
    p->next->prev = temp;
    p->next = temp;
}

void delete_begin()
{
    struct node *last, *p;

    if(start == 0)
    {
        printf("List is empty\n");
        return;
    }

    if(start->next == start)
    {
        free(start);
        start = 0;
        return;
    }

    last = start->prev;
    p = start;

    start = start->next;
    start->prev = last;
    last->next = start;

    free(p);
}

void delete_end()
{
    struct node *last;

    if(start == 0)
    {
        printf("List is empty\n");
        return;
    }

    last = start->prev;

    if(last == start)
    {
        free(start);
        start = 0;
        return;
    }

    last->prev->next = start;
    start->prev = last->prev;

    free(last);
}

void delete_pos()
{
    int pos, i;
    struct node *p;

    if(start == 0)
    {
        printf("List is empty\n");
        return;
    }

    printf("Enter position: ");
    scanf("%d", &pos);

    if(pos < 1)
    {
        printf("Invalid position\n");
        return;
    }

    if(pos == 1)
    {
        delete_begin();
        return;
    }

    p = start;

    for(i = 1; i < pos && p->next != start; i++)
        p = p->next;

    if(i != pos)
    {
        printf("Invalid position\n");
        return;
    }

    p->prev->next = p->next;
    p->next->prev = p->prev;

    free(p);
}

void display()
{
    struct node *p;

    if(start == 0)
    {
        printf("List is empty\n");
        return;
    }

    p = start;
    printf("List: ");

    do
    {
        printf("%s <-> ", p->data);
        p = p->next;
    }
    while(p != start);

    printf("(back to start)\n");
}

void search()
{
    char key[100];
    struct node *p;
    int pos = 1;

    if(start == 0)
    {
        printf("List is empty\n");
        return;
    }

    printf("Enter string to search: ");
    scanf("%99s", key);

    p = start;

    do
    {
        if(strcmp(p->data, key) == 0)
        {
            printf("Found at position %d\n", pos);
            return;
        }

        p = p->next;
        pos++;
    }
    while(p != start);

    printf("String not found\n");
}

int main()
{
    int ch;

    while(1)
    {
        printf("\n1.Insert Begin");
        printf("\n2.Insert Position");
        printf("\n3.Insert End");
        printf("\n4.Delete Begin");
        printf("\n5.Delete End");
        printf("\n6.Delete Position");
        printf("\n7.Is Empty");
        printf("\n8.Is Full");
        printf("\n9.Display");
        printf("\n10.Search");
        printf("\n11.Exit");
        printf("\nEnter choice: ");
        scanf("%d", &ch);

        switch(ch)
        {
            case 1: insert_begin(); break;
            case 2: insert_pos(); break;
            case 3: insert_end(); break;
            case 4: delete_begin(); break;
            case 5: delete_end(); break;
            case 6: delete_pos(); break;

            case 7:
                if(isempty())
                    printf("List is empty\n");
                else
                    printf("List is not empty\n");
                break;

            case 8:
                if(isfull())
                    printf("Memory Full\n");
                else
                    printf("Memory Available\n");
                break;

            case 9: display(); break;
            case 10: search(); break;
            case 11: exit(0);
            default: printf("Invalid choice\n");
        }
    }

    return 0;
}
