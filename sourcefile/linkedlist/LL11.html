Q11. Enter the 2 polynomial equations dynamically, multiply them and display the answer.


#include <stdio.h>

struct poly
{
    int coeff;
    int exp;
};

int main()
{
    struct poly a[20], b[20], c[40];
    int n, m, i, j, k = 0;

    printf("Enter terms in first polynomial: ");
    scanf("%d", &n);

    for(i = 0; i < n; i++)
    {
        printf("Enter coefficient and exponent: ");
        scanf("%d%d", &a[i].coeff, &a[i].exp);
    }

    printf("Enter terms in second polynomial: ");
    scanf("%d", &m);

    for(i = 0; i < m; i++)
    {
        printf("Enter coefficient and exponent: ");
        scanf("%d%d", &b[i].coeff, &b[i].exp);
    }

    i = 0;
    j = 0;

    while(i < n && j < m)
    {
        if(a[i].exp == b[j].exp)
        {
            c[k].coeff = a[i].coeff - b[j].coeff;
            c[k].exp = a[i].exp;
            i++;
            j++;
            k++;
        }
        else if(a[i].exp > b[j].exp)
        {
            c[k++] = a[i++];
        }
        else
        {
            c[k] = b[j];
            c[k].coeff = -b[j].coeff;
            j++;
            k++;
        }
    }

    while(i < n)
        c[k++] = a[i++];

    while(j < m)
    {
        c[k] = b[j];
        c[k].coeff = -b[j].coeff;
        j++;
        k++;
    }

    printf("Difference = ");

    for(i = 0; i < k; i++)
    {
        if(c[i].coeff != 0)
        {
            if(i > 0 && c[i].coeff > 0)
                printf("+ ");

            printf("%dx^%d ", c[i].coeff, c[i].exp);
        }
    }

    printf("\n");

    return 0;
}

Enter coefficient and exponent: 1 2
Enter coefficient and exponent: 4 1
Enter coefficient and exponent: 2 0
