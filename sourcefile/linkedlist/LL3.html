Q3. Implement the operations of doubly linked list storing integer values (insert at begin,insert at pos,insert at end,delete at begin,delete at end, delete at pos, isempty, isfull, display, search)


#include <stdio.h>
#include <stdlib.h>

struct node
{
    int data;
    struct node *prev;
    struct node *next;
};

struct node *start = 0;

int isempty()
{
    return start == 0;
}

int isfull()
{
    struct node *temp;

    temp = (struct node *)malloc(sizeof(struct node));

    if (temp == 0)
        return 1;

    free(temp);
    return 0;
}

void insertbeg()
{
    struct node *temp;

    if (isfull())
    {
        printf("List is full\n");
        return;
    }

    temp = (struct node *)malloc(sizeof(struct node));

    printf("Enter value: ");
    scanf("%d", &temp->data);

    temp->prev = 0;
    temp->next = start;

    if (start != 0)
        start->prev = temp;

    start = temp;

    printf("Node inserted\n");
}

void insertend()
{
    struct node *temp, *p;

    if (isfull())
    {
        printf("List is full\n");
        return;
    }

    temp = (struct node *)malloc(sizeof(struct node));

    printf("Enter value: ");
    scanf("%d", &temp->data);

    temp->next = 0;

    if (start == 0)
    {
        temp->prev = 0;
        start = temp;
    }
    else
    {
        p = start;

        while (p->next != 0)
            p = p->next;

        p->next = temp;
        temp->prev = p;
    }

    printf("Node inserted\n");
}

void insertpos()
{
    struct node *temp, *p;
    int pos, i;

    printf("Enter position: ");
    scanf("%d", &pos);

    if (pos < 1)
    {
        printf("Invalid position\n");
        return;
    }

    if (pos == 1)
    {
        insertbeg();
        return;
    }

    p = start;

    for (i = 1; i < pos - 1 && p != 0; i++)
        p = p->next;

    if (p == 0)
    {
        printf("Invalid position\n");
        return;
    }

    if (isfull())
    {
        printf("List is full\n");
        return;
    }

    temp = (struct node *)malloc(sizeof(struct node));

    printf("Enter value: ");
    scanf("%d", &temp->data);

    temp->next = p->next;
    temp->prev = p;

    if (p->next != 0)
        p->next->prev = temp;

    p->next = temp;

    printf("Node inserted\n");
}

void deletebeg()
{
    struct node *temp;

    if (isempty())
    {
        printf("List is empty\n");
        return;
    }

    temp = start;
    start = start->next;

    if (start != 0)
        start->prev = 0;

    free(temp);

    printf("Node deleted\n");
}

void deleteend()
{
    struct node *p;

    if (isempty())
    {
        printf("List is empty\n");
        return;
    }

    p = start;

    if (p->next == 0)
    {
        start = 0;
    }
    else
    {
        while (p->next != 0)
            p = p->next;

        p->prev->next = 0;
    }

    free(p);

    printf("Node deleted\n");
}

void deletepos()
{
    struct node *p;
    int pos, i;

    printf("Enter position: ");
    scanf("%d", &pos);

    if (isempty() || pos < 1)
    {
        printf("Invalid position or empty list\n");
        return;
    }

    p = start;

    for (i = 1; i < pos && p != 0; i++)
        p = p->next;

    if (p == 0)
    {
        printf("Invalid position\n");
        return;
    }

    if (p->prev != 0)
        p->prev->next = p->next;
    else
        start = p->next;

    if (p->next != 0)
        p->next->prev = p->prev;

    free(p);

    printf("Node deleted\n");
}

void display()
{
    struct node *p;

    if (isempty())
    {
        printf("List is empty\n");
        return;
    }

    p = start;

    printf("Doubly Linked List: ");

    while (p != 0)
    {
        printf("%d <-> ", p->data);
        p = p->next;
    }

    printf("0\n");
}

void search()
{
    struct node *p;
    int key, pos = 1;

    p = start;

    printf("Enter value to search: ");
    scanf("%d", &key);

    while (p != 0)
    {
        if (p->data == key)
        {
            printf("Value found at position %d\n", pos);
            return;
        }

        p = p->next;
        pos++;
    }

    printf("Value not found\n");
}

int main()
{
    int ch;

    while (1)
    {
        printf("\n--- DOUBLY LINKED LIST ---\n");
        printf("1. Insert at Beginning\n");
        printf("2. Insert at Position\n");
        printf("3. Insert at End\n");
        printf("4. Delete at Beginning\n");
        printf("5. Delete at End\n");
        printf("6. Delete at Position\n");
        printf("7. IsEmpty\n");
        printf("8. IsFull\n");
        printf("9. Display\n");
        printf("10. Search\n");
        printf("11. Exit\n");

        printf("Enter choice: ");
        scanf("%d", &ch);

        switch (ch)
        {
            case 1: insertbeg(); break;
            case 2: insertpos(); break;
            case 3: insertend(); break;
            case 4: deletebeg(); break;
            case 5: deleteend(); break;
            case 6: deletepos(); break;

            case 7:
                if (isempty())
                    printf("List is empty\n");
                else
                    printf("List is not empty\n");
                break;

            case 8:
                if (isfull())
                    printf("List is full\n");
                else
                    printf("Memory is available\n");
                break;

            case 9: display(); break;
            case 10: search(); break;
            case 11: return 0;

            default: printf("Invalid choice\n");
        }
    }
}
