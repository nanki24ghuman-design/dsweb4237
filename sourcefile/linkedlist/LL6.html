Q6. Implement the operations of single circular linked list storing string values (insert at begin,insert at pos,insert at end,delete at begin,delete at end, delete at pos, isempty, isfull, display, search)


#include <stdio.h>
#include <stdlib.h>
#include <string.h>

struct node
{
    char data[100];
    struct node *next;
};

struct node *start = 0;

int isempty()
{
    return start == 0;
}

int isfull()
{
    struct node *temp;

    temp = (struct node *)malloc(sizeof(struct node));

    if (temp == 0)
        return 1;

    free(temp);
    return 0;
}

void insertbeg()
{
    struct node *temp, *p;

    if (isfull())
    {
        printf("List is full\n");
        return;
    }

    temp = (struct node *)malloc(sizeof(struct node));

    printf("Enter string: ");
    scanf(" %s", temp->data);

    if (start == 0)
    {
        start = temp;
        temp->next = start;
    }
    else
    {
        p = start;

        while (p->next != start)
            p = p->next;

        temp->next = start;
        p->next = temp;
        start = temp;
    }

    printf("String inserted\n");
}

void insertend()
{
    struct node *temp, *p;

    if (isfull())
    {
        printf("List is full\n");
        return;
    }

    temp = (struct node *)malloc(sizeof(struct node));

    printf("Enter string: ");
    scanf(" %s", temp->data);

    if (start == 0)
    {
        start = temp;
        temp->next = start;
    }
    else
    {
        p = start;

        while (p->next != start)
            p = p->next;

        p->next = temp;
        temp->next = start;
    }

    printf("String inserted\n");
}

void insertpos()
{
    struct node *temp, *p;
    int pos, i;

    printf("Enter position: ");
    scanf("%d", &pos);

    if (pos < 1)
    {
        printf("Invalid position\n");
        return;
    }

    if (pos == 1)
    {
        insertbeg();
        return;
    }

    if (start == 0)
    {
        printf("Invalid position\n");
        return;
    }

    p = start;

    for (i = 1; i < pos - 1 && p->next != start; i++)
        p = p->next;

    if (i != pos - 1)
    {
        printf("Invalid position\n");
        return;
    }

    if (isfull())
    {
        printf("List is full\n");
        return;
    }

    temp = (struct node *)malloc(sizeof(struct node));

    printf("Enter string: ");
    scanf(" %99[\n]", temp->data);

    temp->next = p->next;
    p->next = temp;

    printf("String inserted\n");
}

void deletebeg()
{
    struct node *temp, *p;

    if (isempty())
    {
        printf("List is empty\n");
        return;
    }

    temp = start;

    if (start->next == start)
    {
        start = 0;
    }
    else
    {
        p = start;

        while (p->next != start)
            p = p->next;

        start = start->next;
        p->next = start;
    }

    free(temp);
    printf("String deleted\n");
}

void deleteend()
{
    struct node *p, *temp;

    if (isempty())
    {
        printf("List is empty\n");
        return;
    }

    if (start->next == start)
    {
        free(start);
        start = 0;
    }
    else
    {
        p = start;

        while (p->next->next != start)
            p = p->next;

        temp = p->next;
        p->next = start;

        free(temp);
    }

    printf("String deleted\n");
}

void deletepos()
{
    struct node *p, *temp;
    int pos, i;

    printf("Enter position: ");
    scanf("%d", &pos);

    if (isempty() || pos < 1)
    {
        printf("Invalid position or empty list\n");
        return;
    }

    if (pos == 1)
    {
        deletebeg();
        return;
    }

    p = start;

    for (i = 1; i < pos - 1 && p->next != start; i++)
        p = p->next;

    if (i != pos - 1 || p->next == start)
    {
        printf("Invalid position\n");
        return;
    }

    temp = p->next;
    p->next = temp->next;

    free(temp);
    printf("String deleted\n");
}

void display()
{
    struct node *p;

    if (isempty())
    {
        printf("List is empty\n");
        return;
    }

    p = start;

    printf("Circular Linked List: ");

    do
    {
        printf("%s -> ", p->data);
        p = p->next;
    }
    while (p != start);

    printf("(back to start)\n");
}

void search()
{
    struct node *p;
    char key[100];
    int pos = 1;

    if (isempty())
    {
        printf("List is empty\n");
        return;
    }

    printf("Enter string to search: ");
    scanf(" %99[^\n]", key);

    p = start;

    do
    {
        if (strcmp(p->data, key) == 0)
        {
            printf("String found at position %d\n", pos);
            return;
        }

        p = p->next;
        pos++;
    }
    while (p != start);

    printf("String not found\n");
}

int main()
{
    int ch;

    while (1)
    {
        printf("\n--- CIRCULAR LINKED LIST OF STRINGS ---\n");
        printf("1. Insert at Beginning\n");
        printf("2. Insert at Position\n");
        printf("3. Insert at End\n");
        printf("4. Delete at Beginning\n");
        printf("5. Delete at End\n");
        printf("6. Delete at Position\n");
        printf("7. IsEmpty\n");
        printf("8. IsFull\n");
        printf("9. Display\n");
        printf("10. Search\n");
        printf("11. Exit\n");

        printf("Enter choice: ");
        scanf("%d", &ch);

        switch (ch)
        {
            case 1: insertbeg(); break;
            case 2: insertpos(); break;
            case 3: insertend(); break;
            case 4: deletebeg(); break;
            case 5: deleteend(); break;
            case 6: deletepos(); break;

            case 7:
                if (isempty())
                    printf("List is empty\n");
                else
                    printf("List is not empty\n");
                break;

            case 8:
                if (isfull())
                    printf("List is full\n");
                else
                    printf("Memory is available\n");
                break;

            case 9: display(); break;
            case 10: search(); break;
            case 11: return 0;

            default: printf("Invalid choice\n");
        }
    }
}
