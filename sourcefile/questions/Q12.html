Q12. Implement a dynamic queue using malloc and linked list with enqueue, dequeue, search, display and exit operations.


#include <stdio.h>
#include <stdlib.h>

struct node
{
    int data;
    struct node *next;
};

struct node *front = NULL;
struct node *rear = NULL;

void enqueue()
{
    struct node *temp;
    int value;

    temp = (struct node *)malloc(sizeof(struct node));

    if(temp == NULL)
    {
        printf("Memory is full");
        return;
    }

    printf("Enter value: ");
    scanf("%d", &value);

    temp->data = value;
    temp->next = NULL;

    if(front == NULL)
    {
        front = temp;
        rear = temp;
    }
    else
    {
        rear->next = temp;
        rear = temp;
    }

    printf("Value inserted");
}

void dequeue()
{
    struct node *temp;

    if(front == NULL)
    {
        printf("Queue is empty");
    }
    else
    {
        temp = front;
        printf("Deleted value: %d", temp->data);

        front = front->next;

        if(front == NULL)
        {
            rear = NULL;
        }

        free(temp);
    }
}

void search()
{
    struct node *temp;
    int value, found = 0;

    printf("Enter value to search: ");
    scanf("%d", &value);

    temp = front;

    while(temp != NULL)
    {
        if(temp->data == value)
        {
            found = 1;
            break;
        }

        temp = temp->next;
    }

    if(found == 1)
    {
        printf("Value found");
    }
    else
    {
        printf("Value not found");
    }
}

void display()
{
    struct node *temp;

    if(front == NULL)
    {
        printf("Queue is empty");
    }
    else
    {
        temp = front;

        printf("Queue: ");

        while(temp != NULL)
        {
            printf("%d ", temp->data);
            temp = temp->next;
        }
    }
}

int main()
{
    int choice;

    while(1)
    {
        printf("\n1. Enqueue\n2. Dequeue\n3. Search\n4. Display\n5. Exit\n");
        printf("Enter choice: ");
        scanf("%d", &choice);

        if(choice == 1)
        {
            enqueue();
        }
        else if(choice == 2)
        {
            dequeue();
        }
        else if(choice == 3)
        {
            search();
        }
        else if(choice == 4)
        {
            display();
        }
        else if(choice == 5)
        {
            break;
        }
        else
        {
            printf("Invalid choice");
        }
    }

    return 0;
}

Enter choice: 3
Enter value to search: 20

Enter choice: 2

Enter choice: 5

Complexity

Time Complexity:
Enqueue : O(1)  (rear pointer is kept)
Dequeue : O(1)
Search  : O(n)
Display : O(n)

Space Complexity: O(n)  (one node for each element).
