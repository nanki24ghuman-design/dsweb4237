Q14. Write a C program for a singly linked list with insertion at beginning, end and position, deletion at beginning, end and position, searching, sorting and displaying.


#include <stdio.h>
#include <stdlib.h>

struct node
{
    int data;
    struct node *next;
};

struct node *start = NULL;

void insertbegin()
{
    struct node *temp;

    temp = (struct node *)malloc(sizeof(struct node));

    if(temp == NULL)
    {
        printf("Memory is full");
        return;
    }

    printf("Enter value: ");
    scanf("%d", &temp->data);

    temp->next = start;
    start = temp;
}

void insertend()
{
    struct node *temp, *p;

    temp = (struct node *)malloc(sizeof(struct node));

    if(temp == NULL)
    {
        printf("Memory is full");
        return;
    }

    printf("Enter value: ");
    scanf("%d", &temp->data);

    temp->next = NULL;

    if(start == NULL)
    {
        start = temp;
    }
    else
    {
        p = start;

        while(p->next != NULL)
        {
            p = p->next;
        }

        p->next = temp;
    }
}

void insertposition()
{
    struct node *temp, *p;
    int pos, i;

    printf("Enter position: ");
    scanf("%d", &pos);

    if(pos < 1)
    {
        printf("Invalid position");
        return;
    }

    if(pos == 1)
    {
        insertbegin();
        return;
    }

    p = start;

    for(i = 1; i < pos - 1 && p != NULL; i++)
    {
        p = p->next;
    }

    if(p == NULL)
    {
        printf("Invalid position");
        return;
    }

    temp = (struct node *)malloc(sizeof(struct node));

    if(temp == NULL)
    {
        printf("Memory is full");
        return;
    }

    printf("Enter value: ");
    scanf("%d", &temp->data);

    temp->next = p->next;
    p->next = temp;
}

void deletebegin()
{
    struct node *temp;

    if(start == NULL)
    {
        printf("List is empty");
        return;
    }

    temp = start;
    start = start->next;

    free(temp);

    printf("Node deleted");
}

void deleteend()
{
    struct node *p, *temp;

    if(start == NULL)
    {
        printf("List is empty");
        return;
    }

    if(start->next == NULL)
    {
        free(start);
        start = NULL;
    }
    else
    {
        p = start;

        while(p->next->next != NULL)
        {
            p = p->next;
        }

        temp = p->next;
        p->next = NULL;

        free(temp);
    }

    printf("Node deleted");
}

void deleteposition()
{
    struct node *p, *temp;
    int pos, i;

    printf("Enter position: ");
    scanf("%d", &pos);

    if(pos < 1 || start == NULL)
    {
        printf("Invalid position or empty list");
        return;
    }

    if(pos == 1)
    {
        deletebegin();
        return;
    }

    p = start;

    for(i = 1; i < pos - 1 && p != NULL; i++)
    {
        p = p->next;
    }

    if(p == NULL || p->next == NULL)
    {
        printf("Invalid position");
        return;
    }

    temp = p->next;
    p->next = temp->next;

    free(temp);

    printf("Node deleted");
}

void search()
{
    struct node *p;
    int value, pos = 1;

    printf("Enter value to search: ");
    scanf("%d", &value);

    p = start;

    while(p != NULL)
    {
        if(p->data == value)
        {
            printf("Value found at position %d", pos);
            return;
        }

        p = p->next;
        pos++;
    }

    printf("Value not found");
}

void sort()
{
    struct node *p, *q;
    int temp;

    if(start == NULL)
    {
        printf("List is empty");
        return;
    }

    for(p = start; p->next != NULL; p = p->next)
    {
        for(q = p->next; q != NULL; q = q->next)
        {
            if(p->data > q->data)
            {
                temp = p->data;
                p->data = q->data;
                q->data = temp;
            }
        }
    }

    printf("List sorted");
}

void display()
{
    struct node *p;

    if(start == NULL)
    {
        printf("List is empty");
        return;
    }

    p = start;

    printf("List: ");

    while(p != NULL)
    {
        printf("%d ", p->data);
        p = p->next;
    }
}

int main()
{
    int choice;

    while(1)
    {
        printf("\n\n1. Insert at beginning");
        printf("\n2. Insert at end");
        printf("\n3. Insert at position");
        printf("\n4. Delete at beginning");
        printf("\n5. Delete at end");
        printf("\n6. Delete at position");
        printf("\n7. Search");
        printf("\n8. Sort");
        printf("\n9. Display");
        printf("\n10. Exit");
        printf("\nEnter choice: ");
        scanf("%d", &choice);

        if(choice == 1)
            insertbegin();
        else if(choice == 2)
            insertend();
        else if(choice == 3)
            insertposition();
        else if(choice == 4)
            deletebegin();
        else if(choice == 5)
            deleteend();
        else if(choice == 6)
            deleteposition();
        else if(choice == 7)
            search();
        else if(choice == 8)
            sort();
        else if(choice == 9)
            display();
        else if(choice == 10)
            break;
        else
            printf("Invalid choice");
    }

    return 0;
}

Complexity

Insert at beginning : O(1)
Insert at end       : O(n)  (must travel to the last node)
Insert at position  : O(n)
Delete at beginning : O(1)
Delete at end       : O(n)
Delete at position  : O(n)
Search              : O(n)
Sort                : O(n^2)  (two nested loops, swaps data)
Display             : O(n)

Space Complexity: O(n) for the nodes, O(1) extra space.
(n = number of nodes in the list)
