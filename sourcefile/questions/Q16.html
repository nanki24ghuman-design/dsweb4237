Q16. Write a C program for a circular singly linked list with insertion at beginning, end and position, deletion at beginning, end and position, searching, sorting and displaying.


#include <stdio.h>
#include <stdlib.h>

struct node
{
    int data;
    struct node *next;
};

struct node *start = NULL;

void insertbegin()
{
    struct node *temp, *p;

    temp = (struct node *)malloc(sizeof(struct node));

    if(temp == NULL)
    {
        printf("Memory is full");
        return;
    }

    printf("Enter value: ");
    scanf("%d", &temp->data);

    if(start == NULL)
    {
        start = temp;
        temp->next = start;
    }
    else
    {
        p = start;

        while(p->next != start)
        {
            p = p->next;
        }

        temp->next = start;
        p->next = temp;
        start = temp;
    }
}

void insertend()
{
    struct node *temp, *p;

    temp = (struct node *)malloc(sizeof(struct node));

    if(temp == NULL)
    {
        printf("Memory is full");
        return;
    }

    printf("Enter value: ");
    scanf("%d", &temp->data);

    if(start == NULL)
    {
        start = temp;
        temp->next = start;
    }
    else
    {
        p = start;

        while(p->next != start)
        {
            p = p->next;
        }

        p->next = temp;
        temp->next = start;
    }
}

void insertposition()
{
    struct node *temp, *p;
    int pos, i;

    printf("Enter position: ");
    scanf("%d", &pos);

    if(pos < 1)
    {
        printf("Invalid position");
        return;
    }

    if(pos == 1)
    {
        insertbegin();
        return;
    }

    if(start == NULL)
    {
        printf("Invalid position");
        return;
    }

    p = start;

    for(i = 1; i < pos - 1 && p->next != start; i++)
    {
        p = p->next;
    }

    if(i != pos - 1)
    {
        printf("Invalid position");
        return;
    }

    temp = (struct node *)malloc(sizeof(struct node));

    if(temp == NULL)
    {
        printf("Memory is full");
        return;
    }

    printf("Enter value: ");
    scanf("%d", &temp->data);

    temp->next = p->next;
    p->next = temp;
}

void deletebegin()
{
    struct node *temp, *p;

    if(start == NULL)
    {
        printf("List is empty");
        return;
    }

    temp = start;

    if(start->next == start)
    {
        start = NULL;
    }
    else
    {
        p = start;

        while(p->next != start)
        {
            p = p->next;
        }

        start = start->next;
        p->next = start;
    }

    free(temp);
    printf("Node deleted");
}

void deleteend()
{
    struct node *p, *temp;

    if(start == NULL)
    {
        printf("List is empty");
        return;
    }

    if(start->next == start)
    {
        free(start);
        start = NULL;
    }
    else
    {
        p = start;

        while(p->next->next != start)
        {
            p = p->next;
        }

        temp = p->next;
        p->next = start;

        free(temp);
    }

    printf("Node deleted");
}

void deleteposition()
{
    struct node *p, *temp;
    int pos, i;

    printf("Enter position: ");
    scanf("%d", &pos);

    if(pos < 1 || start == NULL)
    {
        printf("Invalid position or empty list");
        return;
    }

    if(pos == 1)
    {
        deletebegin();
        return;
    }

    p = start;

    for(i = 1; i < pos - 1 && p->next != start; i++)
    {
        p = p->next;
    }

    if(i != pos - 1 || p->next == start)
    {
        printf("Invalid position");
        return;
    }

    temp = p->next;
    p->next = temp->next;

    free(temp);
    printf("Node deleted");
}

void search()
{
    struct node *p;
    int value, pos = 1;

    if(start == NULL)
    {
        printf("List is empty");
        return;
    }

    printf("Enter value to search: ");
    scanf("%d", &value);

    p = start;

    do
    {
        if(p->data == value)
        {
            printf("Value found at position %d", pos);
            return;
        }

        p = p->next;
        pos++;
    }
    while(p != start);

    printf("Value not found");
}

void sort()
{
    struct node *p, *q;
    int temp;

    if(start == NULL)
    {
        printf("List is empty");
        return;
    }

    for(p = start; p->next != start; p = p->next)
    {
        for(q = p->next; q != start; q = q->next)
        {
            if(p->data > q->data)
            {
                temp = p->data;
                p->data = q->data;
                q->data = temp;
            }
        }
    }

    printf("List sorted");
}

void display()
{
    struct node *p;

    if(start == NULL)
    {
        printf("List is empty");
        return;
    }

    p = start;
    printf("List: ");

    do
    {
        printf("%d ", p->data);
        p = p->next;
    }
    while(p != start);
}

int main()
{
    int choice;

    while(1)
    {
        printf("\n\n1. Insert at beginning");
        printf("\n2. Insert at end");
        printf("\n3. Insert at position");
        printf("\n4. Delete at beginning");
        printf("\n5. Delete at end");
        printf("\n6. Delete at position");
        printf("\n7. Search");
        printf("\n8. Sort");
        printf("\n9. Display");
        printf("\n10. Exit");
        printf("\nEnter choice: ");
        scanf("%d", &choice);

        if(choice == 1)
            insertbegin();
        else if(choice == 2)
            insertend();
        else if(choice == 3)
            insertposition();
        else if(choice == 4)
            deletebegin();
        else if(choice == 5)
            deleteend();
        else if(choice == 6)
            deleteposition();
        else if(choice == 7)
            search();
        else if(choice == 8)
            sort();
        else if(choice == 9)
            display();
        else if(choice == 10)
            break;
        else
            printf("Invalid choice");
    }

    return 0;
}

Complexity

Insert at beginning : O(n)  (must travel to the last node to fix its link)
Insert at end       : O(n)
Insert at position  : O(n)
Delete at beginning : O(n)  (must travel to the last node)
Delete at end       : O(n)
Delete at position  : O(n)
Search              : O(n)
Sort                : O(n^2)  (two nested loops, swaps data)
Display             : O(n)

Space Complexity: O(n) for the nodes, O(1) extra space.
(n = number of nodes in the list)
