Q6. Write a C program for insertion sorting.


#include <stdio.h>

int main()
{
    int a[100], n, i, j, temp;

    printf("Enter the size of array: ");
    scanf("%d", &n);

    printf("Enter the array elements: ");

    for(i = 0; i < n; i++)
    {
        scanf("%d", &a[i]);
    }

    for(i = 1; i < n; i++)
    {
        temp = a[i];
        j = i - 1;

        while(j >= 0 && a[j] > temp)
        {
            a[j + 1] = a[j];
            j--;
        }

        a[j + 1] = temp;
    }

    printf("Sorted array: ");

    for(i = 0; i < n; i++)
    {
        printf("%d ", a[i]);
    }

    return 0;
}

Output

Enter the size of array: 5
Enter the array elements: 9 6 8 5 7
Sorted array: 5 6 7 8 9

Complexity

Time Complexity:
Best  : O(n)   (array is already sorted)
Worst : O(n^2) (array is in reverse order)

Space Complexity: O(n) for the array, O(1) extra space (in-place sort).
