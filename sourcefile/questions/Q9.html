Q9. Write a C program to check whether an expression is correct. Use only alphabets and the operators +, *, /, ^. Do not use isalpha().


#include <stdio.h>

int check(char str[], char op[])
{
    int i = 0, j, allowed;
    int needLetter = 1;

    while(str[i] != '\0')
    {
        if(str[i] == ' ' || str[i] == '\n')
        {
            i++;
            continue;
        }

        if(needLetter == 1)
        {
            if((str[i] >= 'A' && str[i] <= 'Z') ||
               (str[i] >= 'a' && str[i] <= 'z'))
            {
                needLetter = 0;
            }
            else
            {
                return 0;
            }
        }
        else
        {
            allowed = 0;

            for(j = 0; op[j] != '\0'; j++)
            {
                if(str[i] == op[j])
                {
                    allowed = 1;
                    break;
                }
            }

            if(allowed == 0)
            {
                return 0;
            }

            needLetter = 1;
        }

        i++;
    }

    if(needLetter == 1)
    {
        return 0;
    }

    return 1;
}

int main()
{
    char str[100];
    char op[] = "+*/^";
    int result;

    printf("Enter expression: ");
    fgets(str, sizeof(str), stdin);

    result = check(str, op);

    if(result == 1)
    {
        printf("Expression is correct");
    }
    else
    {
        printf("Expression is incorrect");
    }

    return 0;
}

Output

Enter expression: A+B*C
Expression is correct

Complexity

Time Complexity: O(n)  (each character is checked once; the operator list has only 4 symbols, so that check is O(1))

Space Complexity: O(n) for the input string, O(1) extra space.
