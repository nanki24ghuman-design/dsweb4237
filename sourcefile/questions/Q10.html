Q10. Implement a simple queue using global variables.


#include <stdio.h>

int queue[5];
int front = -1, rear = -1;

void enqueue()
{
    int value;

    if(rear == 4)
    {
        printf("Queue is full");
    }
    else
    {
        printf("Enter value: ");
        scanf("%d", &value);

        if(front == -1)
        {
            front = 0;
        }

        rear++;
        queue[rear] = value;

        printf("Value inserted");
    }
}

void dequeue()
{
    if(front == -1)
    {
        printf("Queue is empty");
    }
    else
    {
        printf("Deleted value: %d", queue[front]);

        if(front == rear)
        {
            front = -1;
            rear = -1;
        }
        else
        {
            front++;
        }
    }
}

void display()
{
    int i;

    if(front == -1)
    {
        printf("Queue is empty");
    }
    else
    {
        printf("Queue: ");

        for(i = front; i <= rear; i++)
        {
            printf("%d ", queue[i]);
        }
    }
}

int main()
{
    int choice;

    while(1)
    {
        printf("\n1. Enqueue\n2. Dequeue\n3. Display\n4. Exit\n");
        printf("Enter choice: ");
        scanf("%d", &choice);

        if(choice == 1)
        {
            enqueue();
        }
        else if(choice == 2)
        {
            dequeue();
        }
        else if(choice == 3)
        {
            display();
        }
        else if(choice == 4)
        {
            break;
        }
        else
        {
            printf("Invalid choice");
        }
    }

    return 0;
}

Output

Enter choice: 1
Enter value: 10
Value inserted

Enter choice: 1
Enter value: 20
Value inserted

Enter choice: 3
Queue: 10 20

Enter choice: 2
Deleted value: 10

Enter choice: 4

Complexity

Time Complexity:
Enqueue : O(1)
Dequeue : O(1)
Display : O(n)

Space Complexity: O(n) for the queue array.
