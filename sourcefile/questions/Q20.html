Q20. Multiply two polynomial equations using a linked list.


#include <stdio.h>
#include <stdlib.h>

struct node
{
    int coeff;
    int power;
    struct node *next;
};

struct node *insertend(struct node *start, int c, int p)
{
    struct node *temp, *ptr;

    temp = (struct node *)malloc(sizeof(struct node));
    temp->coeff = c;
    temp->power = p;
    temp->next = NULL;

    if(start == NULL)
        return temp;

    ptr = start;

    while(ptr->next != NULL)
        ptr = ptr->next;

    ptr->next = temp;

    return start;
}

struct node *addterm(struct node *start, int c, int p)
{
    struct node *ptr, *prev, *temp;

    if(c == 0)
        return start;

    ptr = start;
    prev = NULL;

    while(ptr != NULL && ptr->power > p)
    {
        prev = ptr;
        ptr = ptr->next;
    }

    if(ptr != NULL && ptr->power == p)
    {
        ptr->coeff = ptr->coeff + c;
        return start;
    }

    temp = (struct node *)malloc(sizeof(struct node));
    temp->coeff = c;
    temp->power = p;
    temp->next = ptr;

    if(prev == NULL)
        start = temp;
    else
        prev->next = temp;

    return start;
}

void display(struct node *start)
{
    struct node *ptr = start;

    if(start == NULL)
    {
        printf("0\n");
        return;
    }

    while(ptr != NULL)
    {
        printf("%dx^%d", ptr->coeff, ptr->power);

        if(ptr->next != NULL)
            printf(" + ");

        ptr = ptr->next;
    }

    printf("\n");
}

int main()
{
    struct node *p1 = NULL, *p2 = NULL, *result = NULL;
    struct node *a, *b;
    int n, i, c, p;

    printf("Enter number of terms in first polynomial: ");
    scanf("%d", &n);

    for(i = 1; i <= n; i++)
    {
        printf("Enter coefficient and power: ");
        scanf("%d%d", &c, &p);
        p1 = insertend(p1, c, p);
    }

    printf("Enter number of terms in second polynomial: ");
    scanf("%d", &n);

    for(i = 1; i <= n; i++)
    {
        printf("Enter coefficient and power: ");
        scanf("%d%d", &c, &p);
        p2 = insertend(p2, c, p);
    }

    a = p1;

    while(a != NULL)
    {
        b = p2;

        while(b != NULL)
        {
            result = addterm(result,
                             a->coeff * b->coeff,
                             a->power + b->power);

            b = b->next;
        }

        a = a->next;
    }

    printf("First polynomial: ");
    display(p1);

    printf("Second polynomial: ");
    display(p2);

    printf("Product of polynomials: ");
    display(result);

    return 0;
}

Complexity

Let m = terms in first polynomial, n = terms in second polynomial.

Time Complexity:
insertend() : O(k)
addterm()   : O(k)  (k = current length of the list)
m x n term multiplications, each calls addterm() on a list of up to m*n terms
Overall (worst case): O(m^2 * n^2)

Space Complexity: O(m*n)  (maximum number of product terms).
