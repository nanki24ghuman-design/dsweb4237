Q21. Divide two polynomial equations using a linked list.


#include <stdio.h>
#include <stdlib.h>

struct node
{
    float coeff;
    int power;
    struct node *next;
};

struct node *insertend(struct node *start, float c, int p)
{
    struct node *temp, *ptr;

    temp = (struct node *)malloc(sizeof(struct node));
    temp->coeff = c;
    temp->power = p;
    temp->next = NULL;

    if(start == NULL)
        return temp;

    ptr = start;

    while(ptr->next != NULL)
        ptr = ptr->next;

    ptr->next = temp;

    return start;
}

struct node *addterm(struct node *start, float c, int p)
{
    struct node *ptr, *prev, *temp;

    if(c > -0.0001 && c < 0.0001)
        return start;

    ptr = start;
    prev = NULL;

    while(ptr != NULL && ptr->power > p)
    {
        prev = ptr;
        ptr = ptr->next;
    }

    if(ptr != NULL && ptr->power == p)
    {
        ptr->coeff = ptr->coeff + c;

        if(ptr->coeff > -0.0001 && ptr->coeff < 0.0001)
        {
            if(prev == NULL)
                start = ptr->next;
            else
                prev->next = ptr->next;

            free(ptr);
        }

        return start;
    }

    temp = (struct node *)malloc(sizeof(struct node));
    temp->coeff = c;
    temp->power = p;
    temp->next = ptr;

    if(prev == NULL)
        start = temp;
    else
        prev->next = temp;

    return start;
}

void display(struct node *start)
{
    struct node *ptr = start;

    if(start == NULL)
    {
        printf("0\n");
        return;
    }

    while(ptr != NULL)
    {
        printf("%.2fx^%d", ptr->coeff, ptr->power);

        if(ptr->next != NULL)
            printf(" + ");

        ptr = ptr->next;
    }

    printf("\n");
}

int main()
{
    struct node *p1 = NULL, *p2 = NULL;
    struct node *quotient = NULL, *remainder = NULL;
    struct node *a, *b, *temp;
    int n, i, p;
    float c, factor;

    printf("Enter number of terms in dividend: ");
    scanf("%d", &n);

    for(i = 1; i <= n; i++)
    {
        printf("Enter coefficient and power: ");
        scanf("%f%d", &c, &p);
        p1 = insertend(p1, c, p);
    }

    printf("Enter number of terms in divisor: ");
    scanf("%d", &n);

    for(i = 1; i <= n; i++)
    {
        printf("Enter coefficient and power: ");
        scanf("%f%d", &c, &p);
        p2 = insertend(p2, c, p);
    }

    a = p1;

    while(a != NULL)
    {
        remainder = addterm(remainder, a->coeff, a->power);
        a = a->next;
    }

    if(p2 == NULL || p2->coeff == 0)
    {
        printf("Invalid divisor.\n");
        return 0;
    }

    while(remainder != NULL &&
          remainder->power >= p2->power)
    {
        factor = remainder->coeff / p2->coeff;
        p = remainder->power - p2->power;

        quotient = addterm(quotient, factor, p);

        b = p2;

        while(b != NULL)
        {
            remainder = addterm(remainder,
                                -factor * b->coeff,
                                p + b->power);

            b = b->next;
        }
    }

    printf("Quotient: ");
    display(quotient);

    printf("Remainder: ");
    display(remainder);

    return 0;
}

Complexity

Let m = terms in dividend, n = terms in divisor,
q = terms in quotient, r = maximum terms in remainder.

Time Complexity:
insertend() and addterm() : O(k) each  (k = current length of the list)
Building the input lists  : O(m^2 + n^2)
Division steps            : about O(q * n * r)
(each of the q steps updates the remainder with n terms of the divisor)

Space Complexity: O(m + n + q + r).
