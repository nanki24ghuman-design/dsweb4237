Q1. Make a program in C to enter 5 data entries and search a value using linear search. Use input, search value, and linear search functions. Display the result only in main.


#include <stdio.h>

void input(int a[])
{
    int i;

    for(i = 0; i < 5; i++)
    {
        scanf("%d", &a[i]);
    }
}

int searchvalue()
{
    int value;
    scanf("%d", &value);
    return value;
}

int linearsearch(int a[], int value)
{
    int i;

    for(i = 0; i < 5; i++)
    {
        if(a[i] == value)
        {
            return i;
        }
    }

    return -1;
}

int main()
{
    int a[5], value, result;

    printf("Enter 5 numbers: ");
    input(a);

    printf("Enter the value to search: ");
    value = searchvalue();

    result = linearsearch(a, value);

    if(result == -1)
    {
        printf("Value not found");
    }
    else
    {
        printf("Value found at position %d", result + 1);
    }

    return 0;
}

Output

Enter 5 numbers: 10 20 30 40 50
Enter the value to search: 30
Value found at position 3

Complexity

Time Complexity:
input()        : O(n)
searchvalue()  : O(1)
linearsearch() : Best O(1) (found at first place), Worst O(n)
Overall        : O(n)

Space Complexity: O(n) for the array, O(1) extra space.
