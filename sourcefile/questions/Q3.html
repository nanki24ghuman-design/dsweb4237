Q3. Enter an array from the user and count the number of occurrences of each number.


#include <stdio.h>

int main()
{
    int a[100], visited[100] = {0};
    int n, i, j, count;

    printf("Enter the size of array: ");
    scanf("%d", &n);

    printf("Enter the array elements: ");

    for(i = 0; i < n; i++)
    {
        scanf("%d", &a[i]);
    }

    for(i = 0; i < n; i++)
    {
        if(visited[i] == 1)
        {
            continue;
        }

        count = 1;

        for(j = i + 1; j < n; j++)
        {
            if(a[i] == a[j])
            {
                count++;
                visited[j] = 1;
            }
        }

        printf("%d occurs %d times\n", a[i], count);
    }

    return 0;
}

Output

Enter the size of array: 6
Enter the array elements: 5 3 5 5 3 6
5 occurs 3 times
3 occurs 2 times
6 occurs 1 times

Complexity

Time Complexity:
Best  : O(n)   (all numbers are same, the inner loop runs only once)
Worst : O(n^2) (all numbers are different, nested loops)

Space Complexity: O(n) (array a[] and visited[]).
