Q18. Add two polynomial equations using a linked list.


#include <stdio.h>
#include <stdlib.h>

struct node
{
    int coeff;
    int power;
    struct node *next;
};

struct node *insertend(struct node *start, int c, int p)
{
    struct node *temp, *ptr;

    temp = (struct node *)malloc(sizeof(struct node));
    temp->coeff = c;
    temp->power = p;
    temp->next = NULL;

    if(start == NULL)
        return temp;

    ptr = start;

    while(ptr->next != NULL)
        ptr = ptr->next;

    ptr->next = temp;

    return start;
}

void display(struct node *start)
{
    struct node *ptr = start;

    while(ptr != NULL)
    {
        printf("%dx^%d", ptr->coeff, ptr->power);

        if(ptr->next != NULL)
            printf(" + ");

        ptr = ptr->next;
    }

    printf("\n");
}

int main()
{
    struct node *p1 = NULL, *p2 = NULL, *result = NULL;
    struct node *a, *b;
    int n, i, c, p, sum;

    printf("Enter number of terms in first polynomial: ");
    scanf("%d", &n);

    for(i = 1; i <= n; i++)
    {
        printf("Enter coefficient and power: ");
        scanf("%d%d", &c, &p);
        p1 = insertend(p1, c, p);
    }

    printf("Enter number of terms in second polynomial: ");
    scanf("%d", &n);

    for(i = 1; i <= n; i++)
    {
        printf("Enter coefficient and power: ");
        scanf("%d%d", &c, &p);
        p2 = insertend(p2, c, p);
    }

    a = p1;
    b = p2;

    while(a != NULL && b != NULL)
    {
        if(a->power == b->power)
        {
            sum = a->coeff + b->coeff;

            if(sum != 0)
                result = insertend(result, sum, a->power);

            a = a->next;
            b = b->next;
        }
        else if(a->power > b->power)
        {
            result = insertend(result, a->coeff, a->power);
            a = a->next;
        }
        else
        {
            result = insertend(result, b->coeff, b->power);
            b = b->next;
        }
    }

    while(a != NULL)
    {
        result = insertend(result, a->coeff, a->power);
        a = a->next;
    }

    while(b != NULL)
    {
        result = insertend(result, b->coeff, b->power);
        b = b->next;
    }

    printf("First polynomial: ");
    display(p1);

    printf("Second polynomial: ");
    display(p2);

    printf("Sum of polynomials: ");
    display(result);

    return 0;
}

Complexity

Let m = terms in first polynomial, n = terms in second polynomial.

Time Complexity:
insertend()  : O(k)  (k = current length of the list)
Building both polynomials : O(m^2 + n^2)
Adding and building result: O((m+n)^2)
Overall      : O((m+n)^2)

Space Complexity: O(m+n)  (result has at most m+n terms).
