Q8. Write a C program for bracket matching using push and pop. Use only round brackets and local variables.


#include <stdio.h>

void push(char stack[], int *top, char ch)
{
    (*top)++;
    stack[*top] = ch;
}

void pop(int *top)
{
    (*top)--;
}

int main()
{
    char str[100], stack[100];
    int top = -1, i = 0, valid = 1;

    printf("Enter brackets: ");
    scanf("%99s", str);

    while(str[i] != '\0')
    {
        if(str[i] == '(')
        {
            push(stack, &top, str[i]);
        }
        else if(str[i] == ')')
        {
            if(top == -1)
            {
                valid = 0;
                break;
            }
            else
            {
                pop(&top);
            }
        }
        else
        {
            valid = 0;
            break;
        }

        i++;
    }

    if(top != -1)
    {
        valid = 0;
    }

    if(valid == 1)
    {
        printf("Brackets are balanced");
    }
    else
    {
        printf("Brackets are not balanced");
    }

    return 0;
}

Output

Enter brackets: (())
Brackets are balanced

Complexity

Time Complexity: O(n)  (the string is read once, push and pop take O(1))

Space Complexity: O(n)  (stack may hold all '(' in the worst case).
