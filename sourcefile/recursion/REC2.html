Q2. Calculate fibonacci of a number using recursion


#include<stdio.h>
int fibonacci(int n){
    if(n==0)
        return 0;
     if(n==1)
        return 1;

        return fibonacci(n-1) + fibonacci(n-2);
    
}
int main(){
    int n;
    printf("enter the no. of terms:");
    scanf("%d",&n);
    
        printf("fibonacci series is ");
        for(int i=0;i<n;i++){
            printf("%d",fibonacci(i));
        }

    
    return 0;
}

Output

enter the no. of terms: 11
fibonacci series is 011235813213455
