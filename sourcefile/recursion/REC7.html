Q7. Enter the number from the user and add all digits of the number using recursion.


#include<stdio.h>
int sumdigits(int n){
    if(n==0){
    return 0;
    }
    return (n%10) +sumdigits(n/10);
}
int main(){
    int n;
    printf("enter the number:");
    scanf("%d",&n);
    if(n<0){
    n=-n;}
    printf("the sum of digits : %d",sumdigits(n));
    return 0;
    
}

enter the number:1349
the sum of digits : 17
