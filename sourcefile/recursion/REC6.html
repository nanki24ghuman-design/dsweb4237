Q6. Find the smallest element in the integer array using recursion.


#include<stdio.h>
int findsmallest(int arr[], int n){
if(n==0)
    return arr[0];

int smallest= findsmallest(arr,n-1);
if(arr[n-1]<smallest)
return arr[n-1];
else
return smallest; 
}
int main(){
    int n, arr[30];
    printf("enter the number of elements in array:");
    scanf("%d",&n);
    for(int i=0;i<n;i++){
        scanf("%d",&arr[i]);
    }
    printf(" smallest number: %d",findsmallest(arr,n));
    return 0;
}
