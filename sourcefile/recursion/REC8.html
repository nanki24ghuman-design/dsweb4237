Q8. Find the arithmetic mean of an integer array using recursion.


#include <stdio.h>

int sumArray(int arr[], int n)
{
    if (n == 0)
        return 0;

    return arr[n - 1] + sumArray(arr, n - 1);
}

int main()
{
    int n, i, arr[100];
    float mean;

    scanf("%d", &n);

    if (n <= 0 || n > 100)
        return 0;

    for (i = 0; i < n; i++)
    {
        scanf("%d", &arr[i]);
    }

    mean = (float)sumArray(arr, n) / n;

    printf("%.2f\n", mean);

    return 0;
}
