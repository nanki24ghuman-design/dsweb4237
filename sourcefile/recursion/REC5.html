Q5. Find the largest element in the integer array using recursion.


#include<stdio.h>
int findlargest(int arr[], int n){
if(n==0)
    return arr[0];

int largest= findlargest(arr,n-1);
if(arr[n-1]>largest)
return arr[n-1];
else
return largest; 
}
int main(){
    int n, arr[30];
    printf("enter the number of elements in array:");
    scanf("%d",&n);
    for(int i=0;i<n;i++){
        scanf("%d",&arr[i]);
    }
    printf(" largest number: %d",findlargest(arr,n));
    return 0;
}
