Q7. Create a tic tac toe game using 2D arrays


#include<stdio.h>
int main()
{
    char a[3][3];
    int i,j,row,col,turn=0;
    for(i=0;i<3;i++)
    {
        for(j=0;j<3;j++)
        {
            a[i][j]=' ';
        }
    }
    while(turn<9)
    {
        printf("\n");
        for(i=0;i<3;i++)
        {
            for(j=0;j<3;j++)
            {
                printf(" %c ",a[i][j]);
                if(j<2)
                    printf("|");
            }
            printf("\n");
            if(i<2)
                printf("---+---+---\n");
        }
        if(turn%2==0)
            printf("Player X, enter row and column (1-3): ");
        else
            printf("Player O, enter row and column (1-3): ");
        scanf("%d%d",&row,&col);
        row--;
        col--;
        if(row<0||row>2||col<0||col>2||a[row][col]!=' ')
        {
            printf("Invalid move! Try again.\n");
            turn--;
        }
        else if(turn%2==0)
            a[row][col]='X';
        else
            a[row][col]='O';
        turn++;
    }
    printf("\nFinal board:\n");
    for(i=0;i<3;i++)
    {
        for(j=0;j<3;j++)
        {
            printf(" %c ",a[i][j]);
            if(j<2)
                printf("|");
        }
        printf("\n");
        if(i<2)
            printf("---+---+---\n");
    }
    return 0;
}
