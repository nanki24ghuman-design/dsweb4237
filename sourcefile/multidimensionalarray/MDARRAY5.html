Q5. Enter a matrix from user and copy it in another matrix of same size


#include<stdio.h>
int main()
{
    int a[2][2],b[2][2],i,j;
    printf("Enter four elements of matrix: ");
    for(i=0;i<2;i++)
    {
        for(j=0;j<2;j++)
        {
            scanf("%d",&a[i][j]);
        }
    }
    for(i=0;i<2;i++)
    {
        for(j=0;j<2;j++)
        {
            b[i][j]=a[i][j];
        }
    }
    printf("Copied matrix:\n");
    for(i=0;i<2;i++)
    {
        for(j=0;j<2;j++)
        {
            printf("%d ",b[i][j]);
        }
        printf("\n");
    }
    return 0;
}

Output

Enter four elements of matrix:
Enter elements:
1  2
3  4

Copied matrix:
1  2
3  4
