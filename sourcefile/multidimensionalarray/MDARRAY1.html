Q1. Enter a matrix of 4 rows and 2 columns from the user and display it in matrix format


#include<stdio.h>
int main()
{
    int arr[4][2],i,j;
    printf("Enter 8 elements: ");
    for(i=0;i<4;i++)
    {
        for(j=0;j<2;j++)
        {
            scanf("%d",&arr[i][j]);
        }
    }
    printf("Matrix is:\n");
    for(i=0;i<4;i++)
    {
        for(j=0;j<2;j++)
        {
            printf("%d ",arr[i][j]);
        }
        printf("\n");
    }
    return 0;
}

Output

Enter 8 elements: 5 6 8 9 10 11 12 13
Matrix is:
5  6
8  9
10 11
12 13
