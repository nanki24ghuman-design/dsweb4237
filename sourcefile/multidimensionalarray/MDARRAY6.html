Q6. Enter a matrix from user, multiply it with a scalar value and display the answer


#include<stdio.h>
int main()
{
    int a[2][2],i,j,n;
    printf("Enter four elements of matrix: ");
    for(i=0;i<2;i++)
    {
        for(j=0;j<2;j++)
        {
            scanf("%d",&a[i][j]);
        }
    }
    printf("Enter scalar value: ");
    scanf("%d",&n);
    for(i=0;i<2;i++)
    {
        for(j=0;j<2;j++)
        {
            a[i][j]=a[i][j]*n;
        }
    }
    printf("Resultant matrix:\n");
    for(i=0;i<2;i++)
    {
        for(j=0;j<2;j++)
        {
            printf("%d ",a[i][j]);
        }
        printf("\n");
    }
    return 0;
}

Output

Enter four elements of matrix: 8 9 7 6
Enter scalar value: 7
Resultant matrix:
56 63
49 42
